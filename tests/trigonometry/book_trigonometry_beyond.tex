Every statement below is true, like those of \texttt{book\_trigonometry.tex}, but the engine does not decide it yet: a parameter beside the variable, an angle written with $\arccos$, or polynomials of a degree too high for its exact arithmetic. It may answer that it cannot decide; it must never call one of them false.

\begin{enumerate}
\item $\displaystyle \forall x \in \mathbb{R}\ \left( \sin\left(6x\right) + \sin\left(2x\right) = 2\cos^{2} x - 1 \iff \exists k \in \mathbb{Z}\ \left( x = \frac{\pi}{24} + \frac{k\pi}{2} \lor x = \frac{5}{24}\pi + \frac{k\pi}{2} \lor x = \frac{\pi}{4} + \frac{k\pi}{2} \right) \right)$
\item $\displaystyle \forall x \in \mathbb{R}\ \left( \frac{\sin\left(4x\right) - \sin\left(6x\right)}{\cos\left(3x\right) + \cos\left(7x\right)} = 1 \iff \exists k \in \mathbb{Z}\ \left( x = \frac{7}{6}\pi + 2k\pi \lor x = \frac{11}{6}\pi + 2k\pi \right) \right)$
\item $\displaystyle \forall x \in \mathbb{R}\ \left( \cos x + \cos\left(3x\right) = 2\sin\left(6x\right)\cos x \iff \exists k \in \mathbb{Z}\ \left( x = \frac{\pi}{16} + \frac{k\pi}{4} \lor x = \frac{\pi}{8} + \frac{k\pi}{2} \lor x = \frac{\pi}{2} + k\pi \right) \right)$
\item $\displaystyle \forall x \in \mathbb{R}\ \left( \cos\left(\frac{x}{2}\right) - 2\cos x - 2 = 0 \iff \exists k \in \mathbb{Z}\ \left( x = -2\arccos\left(\frac{1}{4}\right) + 4k\pi \lor x = 2\arccos\left(\frac{1}{4}\right) + 4k\pi \lor x = \pi + 2k\pi \right) \right)$
\item $\displaystyle \forall x \in \mathbb{R}\ \left( \left|\cos x + 1\right| = 2\left|\cos x\right| \iff \exists k \in \mathbb{Z}\ \left( x = -\arccos\left(-\frac{1}{3}\right) + 2k\pi \lor x = 2k\pi \lor x = \arccos\left(-\frac{1}{3}\right) + 2k\pi \right) \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \exists x \in \mathbb{R}\ \left( 0 < x \le \frac{\pi}{4} \land k\sin x - \cos x - 1 = 0 \right) \iff \left( k \ge 1 + \sqrt{2} \right) \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \exists x \in \mathbb{R}\ \left( 0 < x \le \frac{\pi}{4} \land \sin x - \cos x = k - 2 \right) \iff \left( 1 < k \le 2 \right) \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \exists x \in \mathbb{R}\ \left( -\frac{\pi}{3} < x < \frac{\pi}{2} \land 2\sin x + 2\sqrt{3}\cos x = k \right) \iff \left( 0 < k \le 2 \lor 2 < k \le 4 \right) \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \exists x \in \mathbb{R}\ \left( 0 < x < \frac{\pi}{2} \land 2\cos x - k\sin x + 1 = 0 \right) \iff \left( k > 1 \right) \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \exists x \in \mathbb{R}\ \left( -\frac{\pi}{2} \le x \le \pi \land \cos\left(\frac{3}{2}\pi + x\right) - k\sin x + 2 - 4k = 0 \right) \iff \left( \frac{1}{3} \le k < \frac{1}{2} \lor \frac{1}{2} \le k \le \frac{3}{5} \right) \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \exists x \in \mathbb{R}\ \left( 0 < x < \frac{\pi}{4} \land \sin x \cos x + \cos^{2} x = k - 2 \right) \iff \left( 3 < k \le \frac{5 + \sqrt{2}}{2} \right) \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \exists x \in \mathbb{R}\ \left( 0 < x < \frac{\pi}{2} \land \sin^{2} x - k = k\sin x \cos x \right) \iff \left( 0 < k < 1 \right) \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \exists x \in \mathbb{R}\ \left( 0 < x \le \frac{3}{4}\pi \land \sin^{2} x - \cos^{2} x + 4k\cos x \sin x - k = 0 \right) \iff \left( -1 \le k < 0 \lor k < -1 \lor k \ge 0 \right) \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \exists x \in \mathbb{R}\ \left( -\frac{\pi}{4} \le x \le \frac{\pi}{2} \land k\cos\left(x + \frac{\pi}{4}\right) - 2\sin\left(x + \frac{\pi}{2}\right) = 0 \right) \iff \left( k \ge \sqrt{2} \lor k \le 0 \right) \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \exists x \in \mathbb{R}\ \left( 0 \le x \le \frac{2}{3}\pi \land 4\cos\left(2x\right) - k + 2 = 0 \right) \iff \left( 0 < k \le 6 \lor -2 \le k \le 0 \right) \right)$
\end{enumerate}
