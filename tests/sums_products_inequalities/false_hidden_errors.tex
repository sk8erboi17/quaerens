\title{False identities whose error is hidden in the first values (24)}
\author{}
\date{}
\begin{document}
\maketitle
Each statement is a true identity with one more term on the right. That term is zero for $n = 1, \ldots, 11$ at least, so the two sides agree for the first values of $n$ and differ later. Every statement is false.

\begin{enumerate}
\item $\displaystyle \sum_{k=1}^{n} k = \frac{n\left(n + 1\right)}{2} + \prod_{j=1}^{11} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} k = \frac{n\left(n + 1\right)}{2} + 3 \prod_{j=1}^{15} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} k^{2} = \frac{n\left(n + 1\right)\left(2n + 1\right)}{6} + \prod_{j=1}^{12} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} k^{2} = \frac{n\left(n + 1\right)\left(2n + 1\right)}{6} + \binom{n - 1}{12}$
\item $\displaystyle \sum_{k=1}^{n} k^{3} = \frac{n^{2}\left(n + 1\right)^{2}}{4} + 3 \prod_{j=1}^{15} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} k^{3} = \frac{n^{2}\left(n + 1\right)^{2}}{4} + \prod_{j=1}^{11} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} 2^{k} = 2^{n + 1} - 2 + \binom{n - 1}{12}$
\item $\displaystyle \sum_{k=1}^{n} 2^{k} = 2^{n + 1} - 2 + \prod_{j=1}^{12} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} k \cdot 2^{k} = \left(n - 1\right) 2^{n + 1} + 2 + \prod_{j=1}^{12} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} k \cdot 2^{k} = \left(n - 1\right) 2^{n + 1} + 2 + \binom{n - 1}{12}$
\item $\displaystyle \sum_{k=1}^{n} \frac{1}{k\left(k + 1\right)} = \frac{n}{n + 1} + 3 \prod_{j=1}^{15} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} \frac{1}{k\left(k + 1\right)} = \frac{n}{n + 1} + \prod_{j=1}^{11} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} F_{k} = F_{n + 2} - 1 + \binom{n - 1}{12}$
\item $\displaystyle \sum_{k=1}^{n} F_{k} = F_{n + 2} - 1 + \prod_{j=1}^{12} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} k \cdot k! = \left(n + 1\right)! - 1 + \prod_{j=1}^{11} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} k \cdot k! = \left(n + 1\right)! - 1 + 3 \prod_{j=1}^{15} \left(n - j\right)$
\item $\displaystyle \prod_{k=1}^{n} \left(1 + \frac{1}{k}\right) = n + 1 + 3 \prod_{j=1}^{15} \left(n - j\right)$
\item $\displaystyle \prod_{k=1}^{n} \left(1 + \frac{1}{k}\right) = n + 1 + \prod_{j=1}^{11} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} \left(2k - 1\right) = n^{2} + \binom{n - 1}{12}$
\item $\displaystyle \sum_{k=1}^{n} \left(2k - 1\right) = n^{2} + \prod_{j=1}^{12} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} \left(-1\right)^{k} k = \frac{\left(-1\right)^{n}\left(2n + 1\right) - 1}{4} + \prod_{j=1}^{11} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} \left(-1\right)^{k} k = \frac{\left(-1\right)^{n}\left(2n + 1\right) - 1}{4} + 3 \prod_{j=1}^{15} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} \binom{k}{2} = \binom{n + 1}{3} + \prod_{j=1}^{12} \left(n - j\right)$
\item $\displaystyle \sum_{k=1}^{n} \binom{k}{2} = \binom{n + 1}{3} + \binom{n - 1}{12}$
\end{enumerate}
