$\mathbb{N} = \{1, 2, 3, \ldots\}$, $\mathbb{N}_{0} = \mathbb{N} \cup \{0\}$. Every statement below is false.

\begin{enumerate}
\item $\displaystyle \forall x \in \mathbb{Z}\ \exists y \in \mathbb{Z}\ (x = 2 y)$
\item $\displaystyle \forall x \in \mathbb{Z}\ \exists y \in \mathbb{Z}\ (x^{2} = 4 y + 2)$
\item $\displaystyle \forall x \in \mathbb{Z}\ \exists y \in \mathbb{Z}\ (x^{3} - x = 4 y)$
\item $\displaystyle \forall x \in \mathbb{Z}\ \exists y \in \mathbb{Z}\ (x = 3 y \lor x = 3 y + 1)$
\item $\displaystyle \forall x \in \mathbb{Z}\ \exists y \in \mathbb{Z}\ (x^{2} = 3 y + 2)$
\item $\displaystyle \forall x \in \mathbb{N}\ \exists y \in \mathbb{N}\ (x = 2 y \lor x = 2 y + 1)$
\item $\displaystyle \forall x \in \mathbb{Z}\ \exists y \in \mathbb{Z}\ (x^{2} + x = 4 y)$
\item $\displaystyle \forall x \in \mathbb{Z}\ \exists y \in \mathbb{Z}\ (x^{2} = 4 y \lor x^{2} = 4 y + 3)$
\item $\displaystyle \forall x \in \mathbb{Z}\ \exists y \in \mathbb{N}_{0}\ (x = 2 y \lor x = 2 y + 1)$
\item $\displaystyle \forall x \in \mathbb{Z}\ \exists y \in \mathbb{Z}\ (x^{4} - x = 4 y)$
\item $\displaystyle \exists x \in \mathbb{Z}\ \forall y \in \mathbb{Z}\ (x = y)$
\item $\displaystyle \forall x \in \mathbb{Q}\ \exists y \in \mathbb{Z}\ (x = 2 y \lor x = 2 y + 1)$
\item $\displaystyle \forall x \in \mathbb{Z}\ \exists y \in \mathbb{Z}\ (x^{3} = 9 y \lor x^{3} = 9 y + 1)$
\item $\displaystyle \forall x \in \mathbb{Z}\ \exists y \in \mathbb{Z}\ (x^{2} + x + 1 = 2 y)$
\end{enumerate}
