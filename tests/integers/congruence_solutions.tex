$\mathbb{N} = \{1, 2, 3, \ldots\}$; $a \mid b$: $b$ is $a$ times an integer; $a \equiv b \pmod{m}$: $m \mid a - b$. Each item asks for which integers the condition holds.

\begin{enumerate}
\item $\displaystyle \{x \in \mathbb{Z} \mid 5x \equiv 8 \pmod{36}\}$
\item $\displaystyle \{x \in \mathbb{Z} \mid 24x \equiv 42 \pmod{15}\}$
\item $\displaystyle \{x \in \mathbb{Z} \mid 17x \equiv 3 \pmod{29}\}$
\item $\displaystyle \{x \in \mathbb{Z} \mid 3x \equiv 1 \pmod{5}\}$
\item $\displaystyle \{x \in \mathbb{Z} \mid 15x \equiv 3 \pmod{7}\}$
\item $\displaystyle \{x \in \mathbb{Z} \mid 53x \equiv 7 \pmod{8}\}$
\item $\displaystyle \{x \in \mathbb{Z} \mid 10x \equiv 2 \pmod{6}\}$
\item $\displaystyle \{x \in \mathbb{Z} \mid 8x \equiv 4 \pmod{9}\}$
\item $\displaystyle \{x \in \mathbb{Z} \mid 200x \equiv 10 \pmod{15}\}$
\item $\displaystyle \{x \in \mathbb{Z} \mid 24x \equiv 66 \pmod{30}\}$
\item $\displaystyle \{x \in \mathbb{Z} \mid \begin{cases} x \equiv 2 \pmod{3} \\ x \equiv 5 \pmod{4} \\ x \equiv 3 \pmod{5} \end{cases}\}$
\item $\displaystyle \{a \in \mathbb{Z} \mid \exists b \in \mathbb{Z}\ (ab \equiv 1 \pmod{12})\}$
\item $\displaystyle \{a \in \mathbb{Z} \mid \exists b \in \mathbb{Z}\ (ab \equiv 1 \pmod{7})\}$
\item $\displaystyle \{x \in \mathbb{Z} \mid 2x \equiv 1 \pmod{4}\}$
\item $\displaystyle \{x \in \mathbb{Z} \mid x^{2} \equiv x \pmod{2}\}$
\item $\displaystyle \{x \in \mathbb{N} \mid x^{2} \equiv 1 \pmod{8}\}$
\item $\displaystyle \{x \in \mathbb{Z} : 6 \mid x^{2} + x\}$
\item $\displaystyle \{x \in \mathbb{Z} \mid 4x \equiv 2 \pmod{6} \land 3 \nmid x - 2\}$
\end{enumerate}
