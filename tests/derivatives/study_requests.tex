Each item asks for the real numbers at which the first or second derivative of a function has a sign, or vanishes: the intervals of monotonicity or of concavity of a study of a function, and its stationary points. The engine answers with the solutions, written as in the books. A point where the function has no value is never a solution, even where the formula of its derivative has one; a point where the function is defined but not on both sides is not one either.

\begin{enumerate}
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d}{dx}\left( \frac{x^{2} - 3}{x + 2} \right) > 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d}{dx}\left( x^{4} + 2x^{3} - 2x^{2} \right) < 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d}{dx}\left( \frac{x^{2} - x}{1 + x^{2}} \right) > 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d^{2}}{dx^{2}}\left( \ln(x^{2} + 1) \right) > 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d^{2}}{dx^{2}}\left( x^{3}e^{x} \right) < 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d}{dx}\left( \frac{1}{x\sqrt{1 + x}} \right) > 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d}{dx}\left( \sin(3x) \right) = 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid 0 \le x \le 2\pi \land \frac{d}{dx}\left( \sin x(1 + \cos x) \right) > 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d}{dx}\left( x\ln^{3} x \right) < 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d}{dx}\left( \ln(x^{2} - 1) \right) > 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d}{dx}\left( \ln x \right) < 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d}{dx}\left( \sqrt{x} \right) > 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d}{dx}\left( x e^{x} \right) > 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d}{dx}\left( \arcsin x \right) > 1\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d}{dx}\left( x|x| \right) \ge 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{d}{dx}\left( x \cdot (2 - 3x)^{3} \right) > 0\}$
\end{enumerate}
