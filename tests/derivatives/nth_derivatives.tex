\title{Derivatives of order $n$, and of a fixed order (45)}
\author{}
\date{}
\begin{document}
\maketitle
Each statement gives the derivative of order $n$ of a function of $x$, for every $n$ from the stated value on, or its derivative of a fixed order. All of them are true where the expressions are defined. The letter $e$ is Euler's number; $a$ is a real number.

\begin{enumerate}
\item $\dfrac{d^n}{dx^n}\sin x=\sin\left(x+\dfrac{n\pi}{2}\right)$ per ogni $n\ge0$.
\item $\dfrac{d^n}{dx^n}\cos x=\cos\left(x+\dfrac{n\pi}{2}\right)$ per ogni $n\ge0$.
\item $\dfrac{d^n}{dx^n}\sin^2x=2^{n-1}\sin\left(2x+\dfrac{(n-1)\pi}{2}\right)$ per ogni $n\ge1$.
\item $\dfrac{d^n}{dx^n}\cos^2x=2^{n-1}\cos\left(2x+\dfrac{n\pi}{2}\right)$ per ogni $n\ge1$.
\item $\dfrac{d^n}{dx^n}\left(x\sin x\right)=x\sin\left(x+\dfrac{n\pi}{2}\right)+n\sin\left(x+\dfrac{(n-1)\pi}{2}\right)$ per ogni $n\ge1$.
\item $\dfrac{d^n}{dx^n}\left(xe^{x}\right)=(x+n)e^{x}$ per ogni $n\ge0$.
\item $\dfrac{d^n}{dx^n}\left(x^2e^{x}\right)=\left(x^2+2nx+n(n-1)\right)e^{x}$ per ogni $n\ge0$.
\item $\dfrac{d^n}{dx^n}\ln x=\dfrac{(-1)^{n-1}(n-1)!}{x^n}$ per ogni $n\ge1$ e $x>0$.
\item $\dfrac{d^n}{dx^n}\left(x\ln x\right)=\dfrac{(-1)^n(n-2)!}{x^{n-1}}$ per ogni $n\ge2$ e $x>0$.
\item $\dfrac{d^n}{dx^n}\dfrac{1}{1+x}=\dfrac{(-1)^n\,n!}{(1+x)^{n+1}}$ per ogni $n\ge0$.
\item $\dfrac{d^n}{dx^n}\ln(1+x)=\dfrac{(-1)^{n-1}(n-1)!}{(1+x)^n}$ per ogni $n\ge1$.
\item $\dfrac{d^n}{dx^n}\dfrac{1}{1-x}=\dfrac{n!}{(1-x)^{n+1}}$ per ogni $n\ge0$.
\item $\dfrac{d^n}{dx^n}e^{2x}=2^{n}e^{2x}$ for every integer $n\ge0$.
\item $\dfrac{d^n}{dx^n}e^{-x}=(-1)^{n}e^{-x}$ for every integer $n\ge0$.
\item $\dfrac{d^n}{dx^n}\sin(3x)=3^{n}\sin\left(3x+\dfrac{n\pi}{2}\right)$ for every integer $n\ge0$.
\item $\dfrac{d^n}{dx^n}\left(x\cos x\right)=x\cos\left(x+\dfrac{n\pi}{2}\right)+n\cos\left(x+\dfrac{(n-1)\pi}{2}\right)$ for every integer $n\ge1$.
\item $\dfrac{d^n}{dx^n}\ln(1-x)=-\dfrac{(n-1)!}{(1-x)^{n}}$ for every integer $n\ge1$.
\item $\dfrac{d^n}{dx^n}\dfrac{1}{1+2x}=\dfrac{(-2)^{n}\,n!}{(1+2x)^{n+1}}$ for every integer $n\ge0$.
\item $\dfrac{d^n}{dx^n}\dfrac{x}{1+x}=\dfrac{(-1)^{n+1}\,n!}{(1+x)^{n+1}}$ for every integer $n\ge1$.
\item $\dfrac{d^n}{dx^n}\dfrac{1}{(1-x)^{2}}=\dfrac{(n+1)!}{(1-x)^{n+2}}$ for every integer $n\ge0$.
\item $\dfrac{d^n}{dx^n}3^{x}=\left(\ln 3\right)^{n}3^{x}$ for every integer $n\ge0$.
\item $\dfrac{d^n}{dx^n}\left(\sin x\cos x\right)=2^{n-1}\sin\left(2x+\dfrac{n\pi}{2}\right)$ for every integer $n\ge0$.
\item \[ \frac{d^{n}}{dx^{n}} \left(e^{a x}\right) = a^{n} e^{a x} \]
\item \[ \frac{d^{2}}{dx^{2}} x^{3} = 6x \]
\item \[ \frac{d}{dx} \ln x = \frac{1}{x} \]
\item \[ \frac{d}{dx} \sqrt{x} = \frac{1}{2\sqrt{x}} \]
\item \[ \frac{d}{dx} \arctan x = \frac{1}{1 + x^{2}} \]
\item \[ \frac{d}{dx} \tan x = 1 + \tan^{2} x \]
\item \[ \frac{d^{3}}{dx^{3}} \left(x e^{x}\right) = \left(x + 3\right) e^{x} \]
\item \[ \frac{d^{2}}{dx^{2}} \sin^{2} x = 2\cos\left(2x\right) \]
\item \[ \frac{d}{dx} \sum_{k=0}^{3} x^{k} = 1 + 2x + 3x^{2} \]
\item \[ \frac{d^{2}}{dx^{2}} \sum_{j=0}^{5} x^{j} = \sum_{j=2}^{5} j(j-1) x^{j-2} \]
\item \[ \frac{d^{n}}{dx^{n}} \sum_{j=0}^{3} x^{j} = 0 \] for every integer $n \ge 4$.
\item \[ \frac{d^{n}}{dx^{n}} \left(1+x\right)^{\frac{1}{2}} = \left(\prod_{j=0}^{n-1} \left(\frac{1}{2} - j\right)\right) \left(1+x\right)^{\frac{1}{2} - n} \] for every integer $n \ge 0$.
\item \[ \frac{d}{dx} \cot x = -1 - \cot^{2} x \]
\item \[ \frac{d^{2}}{dx^{2}} \arctan x = -\frac{2x}{\left(1+x^{2}\right)^{2}} \]
\item \[ \frac{d}{dx} \log_{2}\left(x+1\right) = \frac{1}{\left(x+1\right)\ln 2} \]
\item \[ \frac{d^{n}}{dx^{n}} 2^{3x} = \left(3\ln 2\right)^{n} 2^{3x} \] for every integer $n \ge 0$.
\item \[ \frac{d^{3}}{dx^{3}} \tan x = 2\left(1+\tan^{2} x\right)^{2} + 4\tan^{2} x \left(1+\tan^{2} x\right) \]
\item \[ \frac{d^{n}}{dx^{n}} x^{3} = 0 \] for every integer $n \ge 4$.
\item \[ \frac{d^{n}}{dx^{n}} \cos\left(2x+1\right) = 2^{n} \cos\left(2x + 1 + \frac{n\pi}{2}\right) \] for every integer $n \ge 0$.
\item \[ \frac{d^{2}}{dx^{2}} \sqrt[3]{x+2} = -\frac{2}{9 \left(x+2\right)^{\frac{5}{3}}} \]
\item \[ \frac{d^{n}}{dx^{n}} \frac{1}{\left(2x+3\right)^{2}} = \frac{\left(-2\right)^{n} \left(n+1\right)!}{\left(2x+3\right)^{n+2}} \] for every integer $n \ge 0$.
\item \[ \frac{d}{dx} \left(x^{2} \ln x\right) = 2x \ln x + x \]
\item \[ \frac{d^{n}}{dx^{n}} \left(x e^{-x}\right) = \left(-1\right)^{n} \left(x - n\right) e^{-x} \] for every integer $n \ge 0$.
\end{enumerate}
