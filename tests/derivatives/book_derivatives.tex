Every statement below is true: the derivative of a function, of order one to four or of order $n$, is the result printed in an exercise collection (corrected where the printed result is wrong, as an independent check found). The engine must prove them all.

\begin{enumerate}
\item $\displaystyle \frac{d}{dx}\left( x^{2} \cdot 2^{x} \right) = 2^{x} x(2 + x\ln 2)$
\item $\displaystyle \frac{d}{dx}\left( x^{4} \cdot \log_{\frac{2}{3}} x \right) = x^{3}\left(\log_{\frac{2}{3}} x^{4} + \log_{\frac{2}{3}} e\right)$
\item $\displaystyle \frac{d}{dx}\left( (3x^{2} + 5x) \cdot e^{x} \cdot \log_{\frac{1}{2}} x \right) = e^{x}\left[\log_{\frac{1}{2}} x\,(3x^{2} + 11x + 5) + \log_{\frac{1}{2}} e\,(3x + 5)\right]$
\item $\displaystyle \frac{d}{dx}\left( x^{2} \cdot \tan x \right) = \frac{x(\sin 2x + x)}{\cos^{2} x}$
\item $\displaystyle \frac{d}{dx}\left( 7\cot x \cdot \sqrt[3]{x} \right) = \frac{7(\sin 2x - 6x)}{6\sin^{2} x\sqrt[3]{x^{2}}}$
\item $\displaystyle \frac{d}{dx}\left( 2x^{5} \cdot 3\tan x \right) = \frac{3x^{4}(5\sin 2x + 2x)}{\cos^{2} x}$
\item $\displaystyle \frac{d}{dx}\left( (x^{2} + 1) \cdot \arctan x \cdot \frac{1}{x^{2}} \right) = \frac{1}{x}\left[2\arctan x + \frac{1}{x} - \frac{2(x^{2} + 1)\arctan x}{x^{2}}\right]$
\item $\displaystyle \frac{d}{dx}\left( \frac{2x^{2} - 3}{x + 5 - x^{2}} \right) = \frac{2x^{2} + 14x + 3}{(x + 5 - x^{2})^{2}}$
\item $\displaystyle \frac{d}{dx}\left( \frac{x^{-1} + x^{-2}}{x^{-3} + x^{-4}} \right) = 2x$

\item $\displaystyle \frac{d}{dx}\left( (x^{2} - 1)^{x + 1} \right) = (x^{2} - 1)^{x + 1} \cdot \left(\ln(x^{2} - 1) + \frac{2x}{x - 1}\right)$
\item $\displaystyle \frac{d}{dx}\left( x^{\sin x} \right) = \frac{x^{\sin x}}{x} \cdot (\sin x + x\cos x\ln x)$
\item $\displaystyle \frac{d}{dx}\left( (\arctan x)^{1 + x^{2}} \right) = (\arctan x)^{1 + x^{2}} \cdot \left(2x\ln(\arctan x) + \frac{1}{\arctan x}\right)$

\end{enumerate}
