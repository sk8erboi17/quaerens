\maketitle
The first eight items are results printed wrong in the collections (the right ones are in \texttt{book\_partial\_derivatives.tex}); the others are right results changed by little: a number, a sign, an entry of a matrix, a coordinate of the point, the letter of differentiation. None may be proved, and every one is to be called false.

\begin{enumerate}
\item $\displaystyle f(x, y) = \ln\left( 1 + \sqrt{x^{2} + 2y} \right) \implies \nabla f(2, 0) = \left( \frac{2}{3}, \frac{1}{3} \right)$
\item $\displaystyle f(x, y, z) = \frac{x}{(x^{2} + y^{2} + z^{2})^{\frac{3}{2}}} \implies f_{x} = \frac{-x^{2} + y^{2} + z^{2}}{(x^{2} + y^{2} + z^{2})^{\frac{5}{2}}}$
\item $\displaystyle f(x, y) = e^{x + y} (x^{2} + y) \implies f_{y} = e^{x + y} (x^{2} + 2x + 1)$
\item $\displaystyle f(x, y) = x^{3} y^{2} - x^{4} y^{2} - x^{3} y^{3} \implies H_{f}\left( \frac{1}{2}, \frac{1}{3} \right) = \begin{pmatrix} -\frac{1}{9} & -\frac{7}{12} \\ -\frac{7}{12} & \frac{1}{72} \end{pmatrix}$
\item $\displaystyle f(x, y) = x^{4} + y^{4} - 2(x - y)^{2} + 2 \implies H_{f}(0, 0) = \begin{pmatrix} 4 & 4 \\ 4 & 4 \end{pmatrix}$
\item $\displaystyle f(x, y) = 2y^{2} + x^{2} - y \implies \nabla f(0, 4) = 0$
\item $\displaystyle f(x, y) = 2x^{2} - xy - 3y^{2} - 3x + 7y \implies \nabla f\left( -\frac{11}{25}, -\frac{119}{25} \right) = 0$
\item $\displaystyle f(x, y, z) = x^{2} - 2x + 2y^{2} + 3z^{2} \implies H_{f}(1, 0, 0) = \begin{pmatrix} 2 & 0 & 0 \\ 0 & 4 & 0 \\ 0 & 0 & 8 \end{pmatrix}$
\item $\displaystyle f(x, y) = x^{2} y + (3x + 5y)^{2} \implies f_{x} = 2xy + 18x + 15y$
\item $\displaystyle f(x, y) = x^{2} y + (3x + 5y)^{2} \implies H_{f} = \begin{pmatrix} 2y + 18 & 2x + 30 \\ 2x + 30 & 25 \end{pmatrix}$
\item $\displaystyle f(x, y) = \frac{\arctan\frac{x}{y}}{x^{2} + y^{2}} \implies f_{x} = \frac{y + 2x \arctan\frac{x}{y}}{(x^{2} + y^{2})^{2}}$
\item $\displaystyle f(x, y) = \ln(1 + \cos(xy)) \implies \nabla f = \left( -\frac{x \sin(xy)}{1 + \cos(xy)}, -\frac{y \sin(xy)}{1 + \cos(xy)} \right)$
\item $\displaystyle f(x, y) = \sqrt{x^{2} + 3y^{2}} \implies \nabla f = \left( \frac{x}{\sqrt{x^{2} + 3y^{2}}}, \frac{y}{\sqrt{x^{2} + 3y^{2}}} \right)$
\item $\displaystyle f(x, y) = \frac{xy}{x + y} \implies f_{x} = \frac{x^{2}}{(x + y)^{2}}$
\item $\displaystyle f(x, y) = x^{y} \implies f_{yy} = x^{y} \ln x$
\item $\displaystyle f(x, y) = x^{y} \implies H_{f}(1, 2) = \begin{pmatrix} 2 & 1 \\ 1 & 1 \end{pmatrix}$
\item $\displaystyle f(x, y) = x \sin(xy^{2}) \implies f_{xy} = 4xy \cos(xy^{2}) + 2x^{2} y^{3} \sin(xy^{2})$
\item $\displaystyle f(x, y) = \ln(1 - x^{2} - y^{2}) \implies f_{xy} = \frac{4xy}{(1 - x^{2} - y^{2})^{2}}$
\item $\displaystyle f(x, y) = e^{xy} \implies f_{xxy} = y e^{xy} + xy^{2} e^{xy}$
\item $\displaystyle f(x, y, z) = \frac{xyz}{\sqrt{x + y}} + e^{2x - z} \implies \nabla f(1, 0, 2) = (2, 2, 1)$
\item $\displaystyle f(x, y) = xy^{2} + x^{2} y + 2x - y \implies \nabla f(2, 1) = (10, 4)$
\item $\displaystyle f(x, y) = y^{4} e^{3x} \implies \nabla f(0, -1) = (3, 4)$
\item $\displaystyle f(x, y, z) = \frac{1}{x} + \frac{1}{y} + \frac{1}{z} + xyz \implies H_{f}(1, 1, 1) = \begin{pmatrix} 2 & 1 & 1 \\ 1 & 2 & 1 \\ 1 & 1 & -2 \end{pmatrix}$
\item $\displaystyle f(x, y) = \frac{8}{x} + \frac{x}{y} + y \implies \nabla f(2, 4) = 0$
\end{enumerate}
