Each item declares a function of two or three variables and states its partial derivatives, its gradient or its Hessian matrix, as formulas or at a point, or an identity that its derivatives satisfy. Every statement is true where the functions are defined. The sources are listed in \texttt{partial\_derivatives\_about.tex}.

\begin{enumerate}
\item $\displaystyle w(x, y) = x^{3} y - 3xy^{2} + 2y^{2} \implies \frac{\partial w}{\partial x} = 3x^{2} y - 3y^{2} \land \frac{\partial w}{\partial y} = x^{3} - 6xy + 4y$
\item $\displaystyle z(x, y) = \frac{x}{y} \implies \frac{\partial z}{\partial x} = \frac{1}{y} \land \frac{\partial z}{\partial y} = -\frac{x}{y^{2}}$
\item $\displaystyle w(x, y) = \sin(3x + 2y) \implies \nabla w = \left( 3\cos(3x + 2y), 2\cos(3x + 2y) \right)$
\item $\displaystyle w(x, y) = e^{x^{2} y} \implies w_{x} = 2xy e^{x^{2} y} \land w_{y} = x^{2} e^{x^{2} y}$
\item $\displaystyle z(x, y) = x \ln(2x + y) \implies z_{x} = \ln(2x + y) + \frac{2x}{2x + y} \land z_{y} = \frac{x}{2x + y}$
\item $\displaystyle w(x, y, z) = x^{2} z - 2yz^{3} \implies \nabla w = \left( 2xz, -2z^{3}, x^{2} - 6yz^{2} \right)$
\item $\displaystyle f(x, y) = x^{5} y^{3} \implies f_{xy} = f_{yx}$
\item $\displaystyle f(x, y) = \frac{x}{x + y} \implies f_{xy} = f_{yx}$
\item $\displaystyle f(x, y) = \cos(x^{2} + y) \implies \frac{\partial^{2} f}{\partial x \partial y} = \frac{\partial^{2} f}{\partial y \partial x}$
\item $\displaystyle f(x, y) = x^{2} y + 3xy^{2} \implies f_{x} = 2xy + 3y^{2} \land f_{y} = x^{2} + 6xy$
\item $\displaystyle w(x, y) = e^{ax} \sin(ay) \implies w_{xx} + w_{yy} = 0$
\item $\displaystyle w(x, y) = \ln(x^{2} + y^{2}) \implies w_{xx} + w_{yy} = 0$
\item $\displaystyle w(x, y) = \ln(x^{2} + y^{2}) \implies \frac{\partial^{2} w}{\partial x^{2}} = \frac{2y^{2} - 2x^{2}}{(x^{2} + y^{2})^{2}}$
\item $\displaystyle z(x, y) = 3xy + x^{2} \implies \frac{\partial z}{\partial x} = 3y + 2x \land \frac{\partial z}{\partial y} = 3x$
\item $\displaystyle z(x, y) = 3xy + x^{2} \implies \frac{\partial^{2} z}{\partial x^{2}} = 2 \land \frac{\partial^{2} z}{\partial x \partial y} = 3 \land \frac{\partial^{2} z}{\partial y \partial x} = 3 \land \frac{\partial^{2} z}{\partial y^{2}} = 0$
\item $\displaystyle z(x, y) = x^{2} + y^{2} - 2x \implies z_{x} = 2x - 2 \land z_{y} = 2y$
\item $\displaystyle z(x, y) = x^{2} + y^{2} - 2x \implies \nabla z(1, 0) = (0, 0)$
\item $\displaystyle z(x, y) = x^{2} + y^{2} - 2x \implies H_{z} = \begin{pmatrix} 2 & 0 \\ 0 & 2 \end{pmatrix}$
\item $\displaystyle f(x, y) = x^{2} + e^{xy} + 2\ln y \implies \nabla f = \left( 2x + y e^{xy}, x e^{xy} + \frac{2}{y} \right)$
\item $\displaystyle f(x, y) = x^{2} + e^{xy} + 2\ln y \implies H_{f} = \begin{pmatrix} 2 + y^{2} e^{xy} & e^{xy} + xy e^{xy} \\ e^{xy} + xy e^{xy} & x^{2} e^{xy} - \frac{2}{y^{2}} \end{pmatrix}$
\item $\displaystyle f(x, y) = x^{4} + y^{4} - 4xy \implies f_{xx} = 12x^{2} \land f_{xy} = -4 \land f_{yy} = 12y^{2}$
\item $\displaystyle f(x, y) = x^{4} + y^{4} - 4xy \implies \nabla f(1, 1) = 0 \land \nabla f(-1, -1) = 0 \land \nabla f(0, 0) = 0$
\item $\displaystyle f(x, y) = x^{4} + y^{4} - 4xy \implies H_{f}(1, 1) = \begin{pmatrix} 12 & -4 \\ -4 & 12 \end{pmatrix}$
\item $\displaystyle f(x, y) = x^{4} + y^{4} - 4xy \implies H_{f}(0, 0) = \begin{pmatrix} 0 & -4 \\ -4 & 0 \end{pmatrix}$
\item $\displaystyle \frac{\partial}{\partial x}\left( e^{x^{2} y^{3}} \sin(x + z) \right) = 2xy^{3} e^{x^{2} y^{3}} \sin(x + z) + e^{x^{2} y^{3}} \cos(x + z)$
\item $\displaystyle f(x, y) = -\ln(1 - x - y) - \ln x - \ln y \implies \nabla f = \left( \frac{1}{1 - x - y} - \frac{1}{x}, \frac{1}{1 - x - y} - \frac{1}{y} \right)$
\item $\displaystyle f(x, y) = -\ln(1 - x - y) - \ln x - \ln y \implies \nabla f\left( \frac{1}{3}, \frac{1}{3} \right) = (0, 0)$
\item $\displaystyle f(x, y) = -\ln(1 - x - y) - \ln x - \ln y \implies H_{f}\left( \frac{1}{3}, \frac{1}{3} \right) = \begin{pmatrix} 18 & 9 \\ 9 & 18 \end{pmatrix}$
\item $\displaystyle w(x, y, z) = \frac{1}{\sqrt{x^{2} + y^{2} + z^{2}}} \implies w_{xx} + w_{yy} + w_{zz} = 0$
\item $\displaystyle w(x, y) = ax^{2} + bxy - ay^{2} \implies w_{xx} + w_{yy} = 0$
\item $\displaystyle \frac{\partial}{\partial x}\left( \ln(e^{x} + e^{y} + e^{z}) \right) = \frac{e^{x}}{e^{x} + e^{y} + e^{z}}$
\item $\displaystyle f(x, y) = \arctan\left( \frac{y}{x} \right) \implies \frac{\partial f}{\partial x} = -\frac{y}{x^{2} + y^{2}} \land \frac{\partial f}{\partial y} = \frac{x}{x^{2} + y^{2}}$
\item $\displaystyle u(x, y) = x^{2} - y^{2}, v(x, y) = 2xy \implies u_{x} = v_{y} \land u_{y} = -v_{x}$
\item $\displaystyle f(x, y) = x^{3} y^{2} \implies x f_{x} + y f_{y} = 5f$
\item $\displaystyle u(x, t) = e^{-t} \sin x \implies u_{t} = u_{xx}$
\item $\displaystyle u(x, t) = \sin(x - ct) \implies u_{tt} = c^{2} u_{xx}$
\item $\displaystyle f(x, y) = \sqrt{x^{2} + y^{2}} \implies \nabla f = \left( \frac{x}{\sqrt{x^{2} + y^{2}}}, \frac{y}{\sqrt{x^{2} + y^{2}}} \right)$
\item $\displaystyle f(x, y) = \sqrt{x^{2} + y^{2}} \implies \nabla f(3, 4) = \left( \frac{3}{5}, \frac{4}{5} \right)$
\item $\displaystyle f(x, y) = \ln(1 - x - y) \implies f_{x} = \frac{1}{x + y - 1}$
\item $\displaystyle f(x, y) = \sqrt{1 - x^{2} - y^{2}} \implies f_{y} = -\frac{y}{\sqrt{1 - x^{2} - y^{2}}}$
\item $\displaystyle f(x, y) = x^{y} \implies f_{x} = y x^{y - 1} \land f_{y} = x^{y} \ln x$
\item $\displaystyle f(x, y) = x \sin y \implies f_{y}\left( 1, \frac{\pi}{2} \right) = 0 \land f_{x}\left( 1, \frac{\pi}{2} \right) = 1$
\item $\displaystyle f(x, y, z) = xyz + e^{x} \sin z \implies \frac{\partial^{3} f}{\partial x \partial y \partial z} = 1$
\item $\displaystyle f(x, y) = x^{2} + y^{2}, g(x, y) = f(x, y)^{2} \implies g_{x} = 4x(x^{2} + y^{2})$
\item $\displaystyle \frac{\partial}{\partial x}\left( \frac{\partial}{\partial y}\left( x^{2} y^{3} \right) \right) = 6xy^{2}$
\item $\displaystyle f(x, y) = e^{xy} \implies H_{f}(0, 0) = \begin{pmatrix} 0 & 1 \\ 1 & 0 \end{pmatrix}$
\item $\displaystyle f(x, y) = \ln(xy) \implies f_{x}(-1, -1) = -1$
\item $\displaystyle f(x, y) = \ln(x + y) \implies f(1, 1) = \ln 2 \land f_{x}(1, 1) = \frac{1}{2}$
\end{enumerate}
