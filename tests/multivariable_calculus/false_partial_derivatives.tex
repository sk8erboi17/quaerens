Each item is a statement of \texttt{partial\_derivatives.tex} made false by a small change: a coefficient, a sign, the variable of differentiation, a forgotten term, an entry of a matrix, a coordinate of the point. None may be proved, and every one is to be called false.

\begin{enumerate}
\item $\displaystyle w(x, y) = x^{3} y - 3xy^{2} + 2y^{2} \implies \frac{\partial w}{\partial x} = 3x^{2} y - 6y^{2}$
\item $\displaystyle w(x, y) = x^{3} y - 3xy^{2} + 2y^{2} \implies \frac{\partial w}{\partial y} = 3x^{2} y - 3y^{2}$
\item $\displaystyle w(x, y) = x^{3} y - 3xy^{2} + 2y^{2} \implies \frac{\partial w}{\partial y} = x^{3} - 6xy$
\item $\displaystyle z(x, y) = \frac{x}{y} \implies \frac{\partial z}{\partial y} = \frac{x}{y^{2}}$
\item $\displaystyle z(x, y) = \frac{x}{y} \implies \frac{\partial z}{\partial x} = \frac{1}{x}$
\item $\displaystyle w(x, y) = \sin(3x + 2y) \implies \nabla w = \left( 2\cos(3x + 2y), 3\cos(3x + 2y) \right)$
\item $\displaystyle w(x, y) = \sin(3x + 2y) \implies \nabla w = \left( 3\cos(3x + 2y), -2\cos(3x + 2y) \right)$
\item $\displaystyle w(x, y) = e^{x^{2} y} \implies w_{x} = 2x e^{x^{2} y}$
\item $\displaystyle w(x, y) = e^{x^{2} y} \implies w_{y} = 2xy e^{x^{2} y}$
\item $\displaystyle z(x, y) = x \ln(2x + y) \implies z_{x} = \ln(2x + y) + \frac{x}{2x + y}$
\item $\displaystyle w(x, y, z) = x^{2} z - 2yz^{3} \implies \nabla w = \left( 2xz, -2z^{3}, x^{2} - 2yz^{2} \right)$
\item $\displaystyle f(x, y) = x^{5} y^{3} \implies f_{xy} = f_{xx}$
\item $\displaystyle f(x, y) = x^{5} y^{3} \implies f_{xy} = 15x^{4} y^{3}$
\item $\displaystyle f(x, y) = x^{2} y + 3xy^{2} \implies f_{x} = xy + 3y^{2}$
\item $\displaystyle w(x, y) = x^{2} + y^{2} \implies w_{xx} + w_{yy} = 0$

\item $\displaystyle f(x, y) = x^{2} + e^{xy} + 2\ln y \implies \nabla f = \left( 2x + y e^{xy}, x e^{xy} - \frac{2}{y} \right)$
\item $\displaystyle f(x, y) = x^{2} + e^{xy} + 2\ln y \implies H_{f} = \begin{pmatrix} 2 + y^{2} e^{xy} & e^{xy} + xy e^{xy} \\ e^{xy} + xy e^{xy} & x^{2} e^{xy} + \frac{2}{y^{2}} \end{pmatrix}$

\item $\displaystyle f(x, y) = x^{4} + y^{4} - 4xy \implies H_{f}(1, 1) = \begin{pmatrix} 12 & 4 \\ 4 & 12 \end{pmatrix}$

\item $\displaystyle f(x, y) = x^{2} \implies \frac{\partial f}{\partial x} = x$
\item $\displaystyle \frac{\partial}{\partial x}\left( \frac{\partial}{\partial y}\left( x^{2} y^{3} \right) \right) = 6x^{2} y$
\item $\displaystyle f(x, y) = e^{xy} \implies H_{f}(0, 0) = \begin{pmatrix} 0 & 1 \\ 1 & 1 \end{pmatrix}$
\end{enumerate}
