\title{Index notation outside the language (48)}
\author{}
\date{}
\begin{document}
\maketitle
Statements with a Kronecker delta or a Levi-Civita symbol that the reader does not accept: the dimension is not declared, the summation convention is broken, or the notation belongs to a later stage (a metric, derivatives, vectors, powers). Some of them are true in their intended reading; none may be proved or refuted.

\begin{enumerate}
% The dimension is not declared.
\item $\displaystyle \delta_{ii} = 3$
\item $\displaystyle \delta_{ij} a_{j} = a_{i}$
\item $\displaystyle \sum_{i=1}^{3} \delta_{ii} = \varepsilon_{jk}\varepsilon_{jk}$
\item $\displaystyle \varepsilon_{ijk} + \varepsilon_{ij} = 0$
\item $\displaystyle \forall i, j \in \{0, 1, 2, 3\}\ \left( \delta_{ij} = \delta_{ji} \right)$
% The summation convention.
\item $\displaystyle \varepsilon_{ijk} a_{i} b_{i} c_{i} = 0$
\item $\displaystyle (\varepsilon_{ijk} a_{j} b_{k})(\varepsilon_{ijk} c_{j} d_{k}) = a_{j} c_{j} b_{k} d_{k} - a_{j} d_{j} b_{k} c_{k}$
\item $\displaystyle \varepsilon_{ijk} a_{j} b_{k} = c_{j}$
\item $\displaystyle \varepsilon_{ijk} a_{j} b_{k} = a_{i} + b_{j}\varepsilon_{123}$
\item $\displaystyle \forall i \in \{1, 2, 3\}\ \left( \delta_{ij} a_{j} = 1 \right)$
\item $\displaystyle \forall i \in \{1, 2, 3\}\ \left( \delta_{ij} = \delta_{ji} \right)$
\item $\displaystyle \forall i, j \in \{1, 2, 3\}\ \left( \delta_{ij}\delta_{ij} = 3 \right)$
\item $\displaystyle \sum_{i=1}^{3} a_{i}\delta_{jj} = 3 a_{1} + 3 a_{2} + 3 a_{3}$
\item $\displaystyle a_{i}\sum_{j=1}^{3} \delta_{jj} = 3 a_{i}$
% The symbols.
\item $\displaystyle \varepsilon_{124} = 0$
\item $\displaystyle \varepsilon_{012} = 1$
\item $\displaystyle \varepsilon^{123} = 1$
\item $\displaystyle \varepsilon_{ijk}\delta_{ijk} = 0$
\item $\displaystyle \forall i \in \{1, 2, 3\}\ \left( k\,\delta_{ik} = k \right)$
\item $\displaystyle \forall j \in \{1, 2, 3\}\ \left( a_{i}\delta_{ij} = a \right)$
\item $\displaystyle \delta_{ij} a_{i} a_{j}\varepsilon_{123} = a^{2}$
% A metric that is not declared: upper and lower indices that differ, Greek indices.
\item $\displaystyle \varepsilon^{ijk}\varepsilon_{ijk} = 6$
\item $\displaystyle \varepsilon^{ijkl}\varepsilon_{ijkl} = -24$
\item $\displaystyle g_{ij} g^{jk}\varepsilon_{123} = \delta_{i}^{k}$
\item $\displaystyle \varepsilon_{ijk} F^{jk} = \varepsilon_{ijk} F_{jk}$
\item $\displaystyle \varepsilon_{\mu\nu\rho\sigma}\varepsilon^{\mu\nu\rho\sigma} = -24$
\item $\displaystyle \delta^{\mu}_{\mu} = 4$
\item $\displaystyle \sigma_{ij}\delta_{ij}\varepsilon_{123} = \sigma_{kk}$
% Derivatives and Christoffel symbols.
\item $\displaystyle \varepsilon_{ijk}\partial_{i}\partial_{j} f = 0$
\item $\displaystyle \varepsilon_{ijk} A_{j,k} = 0$
\item $\displaystyle \Gamma^{k}_{ij}\varepsilon_{ijk} = 0$
% A declared metric: what is missing, or not allowed.
\item $\displaystyle g_{ij} = \operatorname{diag}(-1, 1, 1, 1) \implies \varepsilon^{ijkl}\varepsilon_{ijkl} = -24$
\item $\displaystyle \varepsilon_{1234} = 1 \implies \varepsilon^{ijkl}\varepsilon_{ijkl} = -24$
\item $\displaystyle g_{ij} = \operatorname{diag}(-1, 1, 1, 1) \land \varepsilon_{123} = 1 \implies \varepsilon^{ijkl}\varepsilon_{ijkl} = -24$
\item $\displaystyle g_{ij} = \operatorname{diag}(-1, 0, 1, 1) \implies g^{ij} g_{ij} = 4$
\item $\displaystyle g_{ij} = \operatorname{diag}(1, r r) \implies g^{ij} g_{ij} = 2$
\item $\displaystyle g_{ij} = \operatorname{diag}(1, 1) \land h_{ij} = \operatorname{diag}(1, 1) \implies g^{ij} h_{ij} = 2$
\item $\displaystyle g_{ij} = \operatorname{diag}(-1, 1, 1, 1) \implies A_{i} B_{i} = B_{i} A_{i}$
\item $\displaystyle g_{ij} = \operatorname{diag}(-1, 1, 1, 1) \implies \delta_{ij} A^{j} = A_{i}$
\item $\displaystyle g_{ij} = \operatorname{diag}(-1, 1, 1, 1) \implies \forall i \in \{1, 2, 3, 4\}\ \left( A_{i} = B^{i} \right)$
\item $\displaystyle g_{ij} = \operatorname{diag}(-1, 1, 1, 1) \implies \forall i \in \{1, 2, 3, 4\}\ \left( A_{i} + B^{i} = B^{i} + A_{i} \right)$
\item $\displaystyle g_{ij} = \operatorname{diag}(2, 3) \implies A_{1} = 2 A^{1}$
% Vectors, powers, other relations, chains.
\item $\displaystyle \varepsilon_{ijk} a_{j} b_{k} = \mathbf{a} \times \mathbf{b}$
\item $\displaystyle \left( \varepsilon_{ijk} a_{j} b_{k} \right)^{2} = 0$
\item $\displaystyle \varepsilon_{ijk}\varepsilon_{ijk} = 3!$
\item $\displaystyle \varepsilon_{ijk} \neq \varepsilon_{jik}$
\item $\displaystyle \forall i, k \in \{1, 2, 3\}\ \left( \delta_{ij}\delta_{jk} \le \delta_{ik} \right)$
\item $\displaystyle \varepsilon_{ijk} = \varepsilon_{jki} = \varepsilon_{kij}$
\end{enumerate}
