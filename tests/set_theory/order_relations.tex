$\mathbb{N} = \{1, 2, 3, \ldots\}$, $\mathbb{N}_{0} = \mathbb{N} \cup \{0\}$; $f(X) = \{f(x) \mid x \in X\}$, $f^{-1}(Y) = \{x \mid f(x) \in Y\}$; $T^{S} = \{f \mid f \colon S \to T\}$; $\exists! y$: there is exactly one $y$.

\begin{enumerate}
\item $\displaystyle R = \{(x, y) \in \mathbb{N}^{2} \mid \exists a \in \mathbb{Z}\ (y = ax)\} \implies (((\forall x \in \mathbb{N}\ ((x, x) \in R) \land \forall x, y\ (((x, y) \in R \land (y, x) \in R) \implies x = y) \land \forall x, y, z\ (((x, y) \in R \land (y, z) \in R) \implies (x, z) \in R))) \land (\lnot (\forall x, y \in \mathbb{N}\ ((x, y) \in R \lor (y, x) \in R))) \land (\forall y \in \mathbb{N}\ ((1, y) \in R)) \land (\lnot (\exists m \in \mathbb{N}\ \forall x \in \mathbb{N}\ ((x, m) \in R))))$
\item $\displaystyle R = \{(x, y) \in \mathbb{N}^{2} \mid \exists n \in \mathbb{N}\ (y = x^{n})\} \implies (((\forall x \in \mathbb{N}\ ((x, x) \in R) \land \forall x, y\ (((x, y) \in R \land (y, x) \in R) \implies x = y) \land \forall x, y, z\ (((x, y) \in R \land (y, z) \in R) \implies (x, z) \in R))) \land (\lnot (\forall x, y \in \mathbb{N}\ ((x, y) \in R \lor (y, x) \in R))) \land (\lnot (\exists m \in \mathbb{N}\ \forall y \in \mathbb{N}\ ((m, y) \in R))) \land (\lnot (\exists m \in \mathbb{N}\ \forall x \in \mathbb{N}\ ((x, m) \in R))))$
\item $\displaystyle R = \{(x, y) \in \mathbb{N}^{2} \mid x = y \lor \{p \in \mathbb{N} \mid p \mid x \land p \neq 1 \land \forall d \in \mathbb{N}\ (d \mid p \implies (d = 1 \lor d = p))\} \subsetneq \{p \in \mathbb{N} \mid p \mid y \land p \neq 1 \land \forall d \in \mathbb{N}\ (d \mid p \implies (d = 1 \lor d = p))\}\} \implies (((\forall x \in \mathbb{N}\ ((x, x) \in R) \land \forall x, y\ (((x, y) \in R \land (y, x) \in R) \implies x = y) \land \forall x, y, z\ (((x, y) \in R \land (y, z) \in R) \implies (x, z) \in R))) \land (\lnot (\forall x, y \in \mathbb{N}\ ((x, y) \in R \lor (y, x) \in R))) \land (\forall y \in \mathbb{N}\ ((1, y) \in R)))$
\item $\displaystyle (S \neq \emptyset \land R = \{(X, Y) \in \mathcal{P}(S)^{2} \mid X = \emptyset \lor X \cap Y = \emptyset\}) \implies (\lnot ((\forall x \in \mathcal{P}(S)\ ((x, x) \in R) \land \forall x, y\ (((x, y) \in R \land (y, x) \in R) \implies x = y) \land \forall x, y, z\ (((x, y) \in R \land (y, z) \in R) \implies (x, z) \in R))))$
\item $\displaystyle R = \{(a + ib, c + id) \mid a, b, c, d \in \mathbb{R} \land (a + ib = c + id \lor a^{2} + b^{2} < c^{2} + d^{2})\} \implies (((\forall x \in \mathbb{C}\ ((x, x) \in R) \land \forall x, y\ (((x, y) \in R \land (y, x) \in R) \implies x = y) \land \forall x, y, z\ (((x, y) \in R \land (y, z) \in R) \implies (x, z) \in R))) \land (\lnot (\forall x, y \in \mathbb{C}\ ((x, y) \in R \lor (y, x) \in R))) \land (\forall z \in \mathbb{C}\ ((0, z) \in R)))$
\item $\displaystyle A = \{a + ib \mid a, b \in \mathbb{Z}\} \land R = \{(a + ib, c + id) \in A^{2} \mid \exists m, n \in \mathbb{N}\ (c = ma \land d = nb)\} \implies ((\forall x \in A\ ((x, x) \in R) \land \forall x, y\ (((x, y) \in R \land (y, x) \in R) \implies x = y) \land \forall x, y, z\ (((x, y) \in R \land (y, z) \in R) \implies (x, z) \in R)))$
\item $\displaystyle ({\le} \subseteq S \times S \land (\forall x \in S\ ((x, x) \in \le) \land \forall x, y\ (((x, y) \in \le \land (y, x) \in \le) \implies x = y) \land \forall x, y, z\ (((x, y) \in \le \land (y, z) \in \le) \implies (x, z) \in \le)) \land \forall x, y \in S\ ((x, y) \in \le \lor (y, x) \in \le) \land X \subseteq S \land X \neq \emptyset \land |X| < \infty) \implies ((\exists m \in X\ \forall x \in X\ (m \le x)) \land (\exists M \in X\ \forall x \in X\ (x \le M)))$
\item $\displaystyle ({\le} \subseteq S \times S \land (\forall x \in S\ ((x, x) \in \le) \land \forall x, y\ (((x, y) \in \le \land (y, x) \in \le) \implies x = y) \land \forall x, y, z\ (((x, y) \in \le \land (y, z) \in \le) \implies (x, z) \in \le)) \land R = \{((x_{1}, y_{1}), (x_{2}, y_{2})) \in (S^{2})^{2} \mid (x_{1} \le x_{2} \land x_{1} \neq x_{2}) \lor (x_{1} = x_{2} \land y_{1} \le y_{2})\}) \implies (((\forall x \in S^{2}\ ((x, x) \in R) \land \forall x, y\ (((x, y) \in R \land (y, x) \in R) \implies x = y) \land \forall x, y, z\ (((x, y) \in R \land (y, z) \in R) \implies (x, z) \in R))) \land ((\forall x, y \in S\ ((x, y) \in \le \lor (y, x) \in \le)) \implies (\forall x, y \in S^{2}\ ((x, y) \in R \lor (y, x) \in R))) \land ((\forall X \subseteq S\ (X \neq \emptyset \implies \exists m \in X\ \forall x \in X\ (m \le x))) \implies (\forall X \subseteq S^{2}\ (X \neq \emptyset \implies \exists m \in X\ \forall x \in X\ ((m, x) \in R)))))$
\item $\displaystyle ((\forall x \in S\ ((x, x) \in \le_{S}) \land \forall x, y\ (((x, y) \in \le_{S} \land (y, x) \in \le_{S}) \implies x = y) \land \forall x, y, z\ (((x, y) \in \le_{S} \land (y, z) \in \le_{S}) \implies (x, z) \in \le_{S})) \land (\forall x \in T\ ((x, x) \in \le_{T}) \land \forall x, y\ (((x, y) \in \le_{T} \land (y, x) \in \le_{T}) \implies x = y) \land \forall x, y, z\ (((x, y) \in \le_{T} \land (y, z) \in \le_{T}) \implies (x, z) \in \le_{T})) \land R = \{((x_{1}, y_{1}), (x_{2}, y_{2})) \in (S \times T)^{2} \mid x_{1} \le_{S} x_{2} \land y_{1} \le_{T} y_{2}\}) \implies ((\forall x \in S \times T\ ((x, x) \in R) \land \forall x, y\ (((x, y) \in R \land (y, x) \in R) \implies x = y) \land \forall x, y, z\ (((x, y) \in R \land (y, z) \in R) \implies (x, z) \in R)))$
\item $\displaystyle \exists S, T\ \exists {\le_{S}}, {\le_{T}}\ (((\forall x \in S\ ((x, x) \in \le_{S}) \land \forall x, y\ (((x, y) \in \le_{S} \land (y, x) \in \le_{S}) \implies x = y) \land \forall x, y, z\ (((x, y) \in \le_{S} \land (y, z) \in \le_{S}) \implies (x, z) \in \le_{S}))) \land (\forall x, y \in S\ ((x, y) \in \le_{S} \lor (y, x) \in \le_{S})) \land ((\forall x \in T\ ((x, x) \in \le_{T}) \land \forall x, y\ (((x, y) \in \le_{T} \land (y, x) \in \le_{T}) \implies x = y) \land \forall x, y, z\ (((x, y) \in \le_{T} \land (y, z) \in \le_{T}) \implies (x, z) \in \le_{T}))) \land (\forall x, y \in T\ ((x, y) \in \le_{T} \lor (y, x) \in \le_{T})) \land (R = \{((x_{1}, y_{1}), (x_{2}, y_{2})) \in (S \times T)^{2} \mid x_{1} \le_{S} x_{2} \land y_{1} \le_{T} y_{2}\}) \land (\lnot (\forall x, y \in S \times T\ ((x, y) \in R \lor (y, x) \in R))))$
\item $\displaystyle ({\le} \subseteq S \times S \land (\forall x \in S\ ((x, x) \in \le) \land \forall x, y\ (((x, y) \in \le \land (y, x) \in \le) \implies x = y) \land \forall x, y, z\ (((x, y) \in \le \land (y, z) \in \le) \implies (x, z) \in \le)) \land A = \mathcal{P}(S) \setminus \{\emptyset\} \land R = \{(X, Y) \in A^{2} \mid X = Y \lor \forall x \in X\ \forall y \in Y\ (x \le y)\}) \implies (((\forall x \in A\ ((x, x) \in R) \land \forall x, y\ (((x, y) \in R \land (y, x) \in R) \implies x = y) \land \forall x, y, z\ (((x, y) \in R \land (y, z) \in R) \implies (x, z) \in R))) \land (\forall x_{0} \in S\ ((\forall x \in S\ (x_{0} \le x)) \implies \forall X \in A\ ((\{x_{0}\}, X) \in R))))$
\item $\displaystyle ({\le} \subseteq S \times S \land (\forall x \in S\ ((x, x) \in \le) \land \forall x, y\ (((x, y) \in \le \land (y, x) \in \le) \implies x = y) \land \forall x, y, z\ (((x, y) \in \le \land (y, z) \in \le) \implies (x, z) \in \le)) \land S \neq \emptyset \land R = \{(f, g) \in (S^{S})^{2} \mid \forall x \in S\ (f(x) \le g(x))\}) \implies (((\forall x \in S^{S}\ ((x, x) \in R) \land \forall x, y\ (((x, y) \in R \land (y, x) \in R) \implies x = y) \land \forall x, y, z\ (((x, y) \in R \land (y, z) \in R) \implies (x, z) \in R))) \land ((\forall x, y \in S^{S}\ ((x, y) \in R \lor (y, x) \in R)) \iff \exists a\ (S = \{a\})))$
\item $\displaystyle (S \neq \emptyset \land T \neq \emptyset \land R = \{(f, g) \in (T^{S})^{2} \mid f = g \lor f(S) \subsetneq g(S)\}) \implies ((\forall x \in T^{S}\ ((x, x) \in R) \land \forall x, y\ (((x, y) \in R \land (y, x) \in R) \implies x = y) \land \forall x, y, z\ (((x, y) \in R \land (y, z) \in R) \implies (x, z) \in R)))$
\item $\displaystyle ({\le} \subseteq S \times S \land (\forall x \in S\ ((x, x) \in \le) \land \forall x, y\ (((x, y) \in \le \land (y, x) \in \le) \implies x = y) \land \forall x, y, z\ (((x, y) \in \le \land (y, z) \in \le) \implies (x, z) \in \le)) \land R \subseteq S \times S \land (\forall x \in S\ ((x, x) \in R) \land \forall x, y\ ((x, y) \in R \implies (y, x) \in R) \land \forall x, y, z\ (((x, y) \in R \land (y, z) \in R) \implies (x, z) \in R)) \land \forall x, y, z \in S\ (((x, z) \in R \land x \le y \land y \le z) \implies (x, y) \in R) \land A = \{\{y \in S \mid (x, y) \in R\} \mid x \in S\} \land G = \{(X, Y) \in A^{2} \mid \forall a \in X\ \exists b \in Y\ (a \le b)\}) \implies ((\forall x \in A\ ((x, x) \in G) \land \forall x, y\ (((x, y) \in G \land (y, x) \in G) \implies x = y) \land \forall x, y, z\ (((x, y) \in G \land (y, z) \in G) \implies (x, z) \in G)))$
\end{enumerate}
