$\mathbb{N} = \{1, 2, 3, \ldots\}$, $\mathbb{N}_{0} = \mathbb{N} \cup \{0\}$; $\exists! y$: there is exactly one $y$. Every statement below is false.

\begin{enumerate}
\item $\displaystyle G = \{(x, y) \in \mathbb{Z}^{2} \mid 2y = x\} \implies (\forall x \in \mathbb{Z}\ \exists! y \in \mathbb{Z}\ ((x, y) \in G))$
\item $\displaystyle G = \{(x, y) \in \mathbb{Z} \times \mathbb{Z} \mid y = 2x + 1\} \implies (\lnot (\forall x \in \mathbb{Z}\ \exists! y \in \mathbb{Z}\ ((x, y) \in G)))$
\item $\displaystyle G = \{(x, y) \in \mathbb{N} \times \mathbb{Z} \mid y^{2} = x\} \implies (\forall x \in \mathbb{N}\ \exists! y \in \mathbb{Z}\ ((x, y) \in G))$
\item $\displaystyle G = \{(x, y) \in \mathbb{N} \times \mathbb{N} \mid y = x - 1\} \implies (\forall x \in \mathbb{N}\ \exists! y \in \mathbb{N}\ ((x, y) \in G))$
\item $\displaystyle G = \{(x, y) \in \mathbb{Z}^{2} \mid x^{2} = y^{2}\} \implies (\forall x \in \mathbb{Z}\ \exists! y \in \mathbb{Z}\ ((x, y) \in G))$
\item $\displaystyle G = \{(x, y) \in \mathbb{N}_{0} \times \mathbb{Z} \mid x^{2} = y^{2}\} \implies (\forall x \in \mathbb{N}_{0}\ \exists! y \in \mathbb{Z}\ ((x, y) \in G))$
\item $\displaystyle R = \{(a, b) \in \mathbb{Z}^{2} \mid a - b \in \mathbb{N}_{0}\} \implies (\forall x, y\ ((x, y) \in R \implies (y, x) \in R))$
\item $\displaystyle R = \{(a, b) \in \mathbb{Z}^{2} \mid a - b \in \mathbb{N}\} \implies (\forall x \in \mathbb{Z}\ ((x, x) \in R))$
\item $\displaystyle R = \{(a, b) \in \mathbb{Z}^{2} \mid a - b \in \mathbb{N}_{0}\} \implies (\lnot (\forall x \in \mathbb{Z}\ ((x, x) \in R)))$
\item $\displaystyle R = \{(a, b) \in \mathbb{Z}^{2} \mid a = b + 1\} \implies (\forall x, y, z\ (((x, y) \in R \land (y, z) \in R) \implies (x, z) \in R))$
\item $\displaystyle R = \{(a, b) \in \mathbb{Z}^{2} \mid a = b + 1\} \implies (\exists x\ ((x, x) \in R))$
\item $\displaystyle R = \{(a, b) \in \mathbb{Z}^{2} \mid a^{2} = b^{2}\} \implies (\lnot (\forall x, y\ ((x, y) \in R \implies (y, x) \in R)))$
\item $\displaystyle R = \{(a, b) \in \mathbb{Z}^{2} \mid a^{2} = b^{2}\} \implies (\forall x, y\ (((x, y) \in R \land (y, x) \in R) \implies x = y))$
\item $\displaystyle R = \{(x, y) \in \mathbb{N} \times \mathbb{Z} \mid x = y^{2}\} \implies ((4, 3) \in R)$
\item $\displaystyle R = \{(x, y) \in \mathbb{N} \times \mathbb{Z} \mid x = y^{2}\} \implies ((0, 0) \in R)$
\item $\displaystyle R = \{(x, y) \in \mathbb{N} \times \mathbb{Z} \mid x = y^{2}\} \implies ((4, -2) \notin R)$
\item $\displaystyle R = \{(x, y) \in \mathbb{N} \times \mathbb{N} \mid x = y\} \implies (\forall x \in \mathbb{Z}\ ((x, x) \in R))$
\item $\displaystyle R = \{(x, y) \in \mathbb{N} \times \mathbb{Z} \mid x = y^{2}\} \implies (\forall x, y\ ((x, y) \in R \implies (y, x) \in R))$
\item $\displaystyle R = \{(x, y) \in \mathbb{Z}^{2} \mid \exists k \in \mathbb{Z}\ (y = kx)\} \implies (\forall k \in \mathbb{Z}\ ((k, 1) \in R))$
\item $\displaystyle R = \{(x, y) \in \mathbb{Z}^{2} \mid \exists k \in \mathbb{Z}\ (y = kx)\} \implies (\forall y \in \mathbb{Z}\ ((2, y) \in R))$
\item $\displaystyle R = \{(x, y) \in \mathbb{Z}^{2} \mid \exists k \in \mathbb{Z}\ (y = kx)\} \implies (\forall k \in \mathbb{Z}\ ((k, k + 1) \in R))$
\item $\displaystyle R = \{(x, y) \in \mathbb{N}^{2} \mid \exists a \in \mathbb{Z}\ (y = ax)\} \implies (\forall x, y \in \mathbb{N}\ ((x, y) \in R \lor (y, x) \in R))$
\item $\displaystyle R = \{(x, y) \in \mathbb{Z}^{2} \mid y = 2x\} \implies (\forall a \in \mathbb{Z}\ ((a + 1, 2a + 1) \in R))$
\item $\displaystyle R = \{(x, y) \in \mathbb{Z}^{2} \mid y = x^{2}\} \implies (\forall a \in \mathbb{Z}\ ((a - 1, a^{2} - 1) \in R))$
\item $\displaystyle R = \{(x, y) \in \mathbb{Z}^{2} \mid y = x^{2}\} \implies (\forall a \in \mathbb{Z}\ ((-a, -a^{2}) \in R))$
\item $\displaystyle (R = \{(x, y) \in \mathbb{Z}^{2} \mid y = 2x\} \land S = \{(x, y) \in \mathbb{Z}^{2} \mid (y, x) \in R\}) \implies (\forall a \in \mathbb{Z}\ ((a, 2a) \in S))$
\item $\displaystyle (R = \{(x, y) \in \mathbb{Z}^{2} \mid y = 2x\} \land S = \{(x, y) \in \mathbb{Z}^{2} \mid (y, x) \in R\}) \implies (\forall x \in \mathbb{Z}\ \exists! y \in \mathbb{Z}\ ((x, y) \in S))$
\item $\displaystyle \forall x \in \mathbb{Z}\ \exists! y \in \mathbb{Z}\ (y^{2} = x^{2})$
\item $\displaystyle \forall x \in \mathbb{N}\ \exists! y \in \mathbb{N}\ (x + y = 5)$
\item $\displaystyle \lnot (\forall x \in \mathbb{Z}\ \exists! y \in \mathbb{Z}\ (x + y = 0))$
\item $\displaystyle \exists! y \in \mathbb{Z}\ (y^{2} = 4)$
\item $\displaystyle \exists! y \in \mathbb{N}\ (y^{2} = 5)$
\item $\displaystyle \forall x \in \mathbb{Z}\ (\frac{x^{2} + x}{3} \in \mathbb{Z})$
\item $\displaystyle \forall x \in \mathbb{Z}\ (x^{3} \in \mathbb{N}_{0})$
\item $\displaystyle \forall n \in \mathbb{N}\ (n - 2 \in \mathbb{N}_{0})$
\item $\displaystyle \exists n \in \mathbb{N}\ (n \notin \mathbb{Z})$
\item $\displaystyle R = \{(x, y) \in \mathbb{Z}^{2} \mid x^{2} = y^{2}\} \implies (\forall x, y, z, w\ (((x, y) \in R \land (z, w) \in R) \implies (x + z, y + w) \in R))$
\item $\displaystyle R = \{(x, y) \in \mathbb{Z}^{2} \mid x^{2} = y^{2}\} \implies (\forall x, y\ ((x, y) \in R \implies x = y))$
\item $\displaystyle R = \{(Z, y) \in \mathbb{Z}^{2} \mid \exists k \in \mathbb{Z}\ (Z + k = y)\} \implies (\forall N \in \mathbb{N}\ ((N, 0) \notin R))$
\end{enumerate}
