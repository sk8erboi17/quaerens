Every statement below is true. Each says that an inequality in the real variable $x$, with a real parameter, holds exactly in the cases printed as its result in a school textbook (exercises 215--223, 225 and 227--231); the last two say for which values of the parameter an inequality holds for every $x$, or for none (exercises 232 and 233). A comparison counts as false where one of its sides has no value.

\begin{enumerate}
\item $\displaystyle \forall a \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( x^{2} - 4ax \le 0 \iff \left( (a < 0 \land 4a \le x \le 0) \lor (a = 0 \land x = 0) \lor (a > 0 \land 0 \le x \le 4a) \right) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( a^{2}x^{2} + 3ax - 10 \ge 0 \iff \left( (a < 0 \land (x \le \frac{2}{a} \lor x \ge -\frac{5}{a})) \lor (a > 0 \land (x \le -\frac{5}{a} \lor x \ge \frac{2}{a})) \right) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( x^{2} + 2ax - 8a^{2} \le 0 \iff \left( (a < 0 \land 2a \le x \le -4a) \lor (a = 0 \land x = 0) \lor (a > 0 \land -4a \le x \le 2a) \right) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( 2a^{2}x^{2} + 5ax - 3 < 0 \iff \left( (a < 0 \land \frac{1}{2a} < x < -\frac{3}{a}) \lor a = 0 \lor (a > 0 \land -\frac{3}{a} < x < \frac{1}{2a}) \right) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( ax^{2} + x(a^{2} - 1) - a < 0 \iff \left( (a < 0 \land (x < \frac{1}{a} \lor x > -a)) \lor (a = 0 \land x > 0) \lor (a > 0 \land -a < x < \frac{1}{a}) \right) \right)$
\item $\displaystyle \forall b \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( (b + 1)x^{2} - 4bx + 4b - 4 > 0 \iff \left( (b < -1 \land 2 < x < \frac{2(b - 1)}{b + 1}) \lor (b = -1 \land x > 2) \lor (b > -1 \land (x < \frac{2(b - 1)}{b + 1} \lor x > 2)) \right) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( -3x^{2} + 2a(a + x) - 3ax \ge 0 \iff \left( (a < 0 \land \frac{2a}{3} \le x \le -a) \lor (a = 0 \land x = 0) \lor (a > 0 \land -a \le x \le \frac{2a}{3}) \right) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( x^{2} - 4 + a(4 - x) < 3x \iff \left( (a < 5 \land a - 1 < x < 4) \lor (a > 5 \land 4 < x < a - 1) \right) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( 3x^{2} + a(1 - 2x) \ge 1 - x^{2} \iff \left( (a < 2 \land (x \le \frac{a - 1}{2} \lor x \ge \frac{1}{2})) \lor a = 2 \lor (a > 2 \land (x \le \frac{1}{2} \lor x \ge \frac{a - 1}{2})) \right) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( ax^{2} + (3 - a^{2})x - 3a \ge 0 \iff \left( (a < 0 \land a \le x \le -\frac{3}{a}) \lor (a = 0 \land x \ge 0) \lor (a > 0 \land (x \le -\frac{3}{a} \lor x \ge a)) \right) \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( x(x + 4) - k(k + 6) - 5 > 0 \iff \left( (k < -3 \land (x < k + 1 \lor x > -k - 5)) \lor (k = -3 \land x \neq -2) \lor (k > -3 \land (x < -k - 5 \lor x > k + 1)) \right) \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( 2x(x - 1) - k(x + 2) + 3k \ge 0 \iff \left( (k < 2 \land (x \le \frac{k}{2} \lor x \ge 1)) \lor k = 2 \lor (k > 2 \land (x \le 1 \lor x \ge \frac{k}{2})) \right) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( x^{2} + 4ax + 5a^{2} > 0 \iff \left( a \neq 0 \lor (a = 0 \land x \neq 0) \right) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( x^{2} - (a + 1)x + a < 0 \iff \left( (a < 1 \land a < x < 1) \lor (a > 1 \land 1 < x < a) \right) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( x(x - a) \le 2(a - x) \iff \left( (a < -2 \land a \le x \le -2) \lor (a = -2 \land x = -2) \lor (a > -2 \land -2 \le x \le a) \right) \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \left(\forall x \in \mathbb{R}\ (x^{2} - 2kx + 7 > 0)\right) \iff \left( -\sqrt{7} < k < \sqrt{7} \right) \right)$
\item $\displaystyle \forall b \in \mathbb{R}\ \left( \left(\lnot \exists x \in \mathbb{R}\ (x^{2} + 3bx - b \le 0)\right) \iff \left( -\frac{4}{9} < b < 0 \right) \right)$
\end{enumerate}
