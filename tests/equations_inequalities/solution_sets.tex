Each item asks for the set of the real numbers that satisfy a condition. Nothing is claimed: the engine must find the solutions. The expected solutions are in \texttt{test\_equations\_inequalities.c}.

\begin{enumerate}
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{x - 2}{x + 3} \ge 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \left|2x + 3\right| - \left|3x - 2\right| < 4\}$
\item $\displaystyle \{x \in \mathbb{R} : x^{2} < 2\}$
\item $\displaystyle \{x \in \mathbb{R} \mid x^{2} + 1 < 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid x^{2} + 1 > 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid (4x + 1)^{2}(2x^{2} - 7x - 4) \ge 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid x^{3} - x - 1 > 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \begin{cases} \frac{1 + x^{2}}{3x} \le 0 \\ x < (x + 2)(3 - x) \end{cases}\}$
\item $\displaystyle \{x \in \mathbb{R} \mid x^{2} - 2x - 1 \le 0 \lor 3x^{2} = 2\}$
\item $\displaystyle \{x \in \mathbb{R}^{+} \mid x^{2} \le 4\}$
\item $\displaystyle \{x \in \mathbb{R} \setminus \{1\} \mid x^{2} \le 4\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{x - 2}{x - 3} \le \frac{9}{\left|x^{2} - 5x + 6\right|}\}$
\item $\displaystyle \{x \in \mathbb{R} \mid -2x^{2} + 4x + 1 = 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{1}{x} \le x\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \left|x\right| = 3\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \left|\left|x\right| - 2\right| \le 1\}$
\item $\displaystyle \{x \in \mathbb{R} \mid (x^{2} - 2)(x^{2} - 3) \le 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid (x^{3} - 2)(x^{3} - 3) < 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid x^{2}(1 - x) + x^{2}(x - 2)(x + 2) - x^{2} - (x - 1)(x - 2) + 4 < 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \begin{cases} \frac{1 + x^{2}}{3x} \le 0 \\ x < (x + 2)(3 - x) \end{cases}\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \begin{cases} \frac{x^{2} - 1}{2} \ge \frac{2}{x^{2} - 1} \\ \frac{x^{2} + 1}{2} \ge \frac{2}{x^{2} + 1} \end{cases}\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \left|x^{2} - 2x\right| \le 2\left|x\right| - 3\}$
\item $\displaystyle \{x \in \mathbb{R} \mid (1 + \left|x\right|)^{2} - 2x - 3 \le 0\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \left|\frac{x^{2} - 2x}{x + 1}\right| < 2\}$
\end{enumerate}
