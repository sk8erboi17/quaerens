Every statement below is true, and its verdict rests on all the cases on the parameter. The critical values of the parameter are far beyond $|a| = 8$ ($100$ and $101$, $-60$ and $12$, $-35$, $-38$, $-30$, $50$ and $51$), or isolated irrational numbers ($\pm\sqrt{79}$), and several statements have degree three or four in $x$. The reason why each one holds is written in the companion file \texttt{parameter\_statements\_beyond\_grid\_about.tex}.

\begin{enumerate}
\item $\displaystyle \forall a \in \mathbb{R}\ \left( \left( \exists x \in \mathbb{R}\ (x^{2} - 2ax + 201a - 10100 < 0) \right) \iff (a < 100 \lor a > 101) \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \left( \forall x \in \mathbb{R}\ (x^{2} - 2kx + 79 > 0) \right) \iff -\sqrt{79} < k < \sqrt{79} \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \left( \exists x \in \mathbb{R}\ (x^{2} - 2kx + 79 = 0) \right) \iff (k \le -\sqrt{79} \lor k \ge \sqrt{79}) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \left( \left( \forall x \in \mathbb{R}\ (x^{4} - 2ax^{2} + 2a + 120 > 0) \right) \iff -60 < a < 12 \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \left( \left( \exists x \in \mathbb{R}\ (x^{4} - 2ax^{2} + 2a + 120 \le 0) \right) \iff (a \le -60 \lor a \ge 12) \right)$
\item $\displaystyle \lnot \exists a \in \mathbb{R}\ \forall x \in \mathbb{R}\ (x^{4} - 2ax^{2} + 2a + 120 < 0)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \left( \forall x \in \mathbb{R}\ (x^{4} - 2kx^{2} + 79 > 0) \right) \iff k < \sqrt{79} \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \forall x \in \mathbb{R}\ (x^{3} - 2ax^{2} - 30x^{2} + x - 2a - 30 > 0 \iff x > 2a + 30)$
\item $\displaystyle \forall a \in \mathbb{R}\ \left( \left( \exists x \in \mathbb{R}\ (x^{3} - ax^{2} - 40x^{2} + 2x - 2a - 80 = 0 \land x > 5) \right) \iff a > -35 \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \left( \left( \forall x \in \mathbb{R}\ (x \ge 2 \lor x^{3} - ax^{2} - 40x^{2} + 2x - 2a - 80 < 0) \right) \iff a \ge -38 \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \forall x \in \mathbb{R}\ \left( \frac{x - a}{x - 2a - 30} \le 0 \iff \left( (a < -30 \land 2a + 30 < x \le a) \lor (a > -30 \land a \le x < 2a + 30) \right) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \left( \left( \forall x \in \mathbb{R}\ (ax^{2} - 50x^{2} - 2x + 1 > 0) \right) \iff a > 51 \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \left( \left( \exists x \in \mathbb{R}\ (ax^{2} - 50x^{2} - 2x + 1 = 0) \right) \iff a \le 51 \right)$
\end{enumerate}
