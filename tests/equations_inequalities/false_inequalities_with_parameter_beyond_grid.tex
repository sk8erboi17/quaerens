Every statement below is false. Each one is a statement of \texttt{parameter\_statements\_beyond\_grid.tex} spoiled a little: an endpoint moved, $<$ written for $\le$ or the other way round, a case forgotten; or a quantifier changed. The value of the parameter where it fails is far beyond $|a| = 8$, or an isolated irrational number, or comes from a polynomial of degree three or four in $x$: it is written, with the reason, in the companion file \texttt{false\_inequalities\_with\_parameter\_beyond\_grid\_about.tex}.

\begin{enumerate}
\item $\displaystyle \forall a \in \mathbb{R}\ \left( \left( \exists x \in \mathbb{R}\ (x^{2} - 2ax + 201a - 10100 < 0) \right) \iff (a \le 100 \lor a > 101) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \left( \left( \exists x \in \mathbb{R}\ (x^{2} - 2ax + 201a - 10100 < 0) \right) \iff (a < 100 \lor a > 102) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \left( \left( \exists x \in \mathbb{R}\ (x^{2} - 2ax + 201a - 10100 < 0) \right) \iff (a < 99 \lor a > 101) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \exists x \in \mathbb{R}\ (x^{2} - 2ax + 201a - 10100 < 0)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \left( \forall x \in \mathbb{R}\ (x^{2} - 2kx + 79 > 0) \right) \iff -\sqrt{79} \le k < \sqrt{79} \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \left( \forall x \in \mathbb{R}\ (x^{2} - 2kx + 79 > 0) \right) \iff -\sqrt{79} < k < \sqrt{80} \right)$
\item $\displaystyle \forall k \in \mathbb{R}\ \left( \left( \exists x \in \mathbb{R}\ (x^{2} - 2kx + 79 = 0) \right) \iff (k < -\sqrt{79} \lor k \ge \sqrt{79}) \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \left( \left( \forall x \in \mathbb{R}\ (x^{4} - 2ax^{2} + 2a + 120 > 0) \right) \iff -60 < a \le 12 \right)$
\item $\displaystyle \forall a \in \mathbb{R}\ \left( \left( \forall x \in \mathbb{R}\ (x^{4} - 2ax^{2} + 2a + 120 > 0) \right) \iff -60 \le a < 12 \right)$
