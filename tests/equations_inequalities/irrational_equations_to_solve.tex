\maketitle
Each item asks for the real numbers that satisfy an equation with the unknown under a square root. Nothing is claimed: the engine must find the solutions. The expected solutions are in \texttt{test\_irrational\_equations.c}.

\begin{enumerate}
\item $\displaystyle \{x \in \mathbb{R} \mid \sqrt{x} - 3 = \sqrt{3} - \sqrt{12 - x}\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{2\sqrt{x + 3} - \sqrt{2 - x}}{\sqrt{x + 8}} = 1\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \sqrt{2x - 1} = 2\sqrt{x + 4} - 3\}$
\item $\displaystyle \{x \in \mathbb{R} \mid -\sqrt{x^{2} - 1} + \sqrt{x^{2} + 3x} = 2\}$
\item $\displaystyle \{x \in \mathbb{R} \mid 2\sqrt{x + 5} - \sqrt{2x + 1} = \sqrt{3x - 3}\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \sqrt{x + 3} = \sqrt{x + 12} - \sqrt{3 - x}\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \sqrt{3} - \sqrt{\frac{x}{4} + 1} = \frac{1}{2}\sqrt{2 - x}\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \sqrt{3x + 13} - \sqrt{3(x + 2)} = \sqrt{x + 3} - \sqrt{x}\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \frac{2}{\sqrt{4x + 5}} = \frac{9}{\sqrt{4x^{2} + x - 5}} - \frac{1}{\sqrt{x - 1}}\}$
\item $\displaystyle \{x \in \mathbb{R} \mid \sqrt{5 - x} + \sqrt{5 + x} = \frac{12}{\sqrt{5 + x}}\}$
\end{enumerate}
