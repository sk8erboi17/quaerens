\item $\displaystyle \{x \in \mathbb{R} \mid \frac{1}{x} \le x\}$
\item $\displaystyle \{x \in \mathbb{R} \mid x^{2} < 2\}$
\item $\displaystyle \forall x \in \mathbb{R}\ \left(\frac{1}{x} > 0 \iff x \ge 0\right)$

\end{enumerate}
