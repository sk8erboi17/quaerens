\maketitle
Every statement below is true. Each says that an equation in the real variable $x$, with $x$ under one or more square roots, holds exactly for the values printed as its result in a school textbook. A comparison counts as false where one of its sides has no value. The engine must prove them all.

\begin{enumerate}
\item $\displaystyle \forall x \in \mathbb{R}\ \left( \sqrt{x} - 3 = \sqrt{3} - \sqrt{12 - x} \iff \left( x = 3 \lor x = 9 \right) \right)$
\item $\displaystyle \forall x \in \mathbb{R}\ \left( \frac{2\sqrt{x + 3} - \sqrt{2 - x}}{\sqrt{x + 8}} = 1 \iff \left( x = 1 \right) \right)$
\item $\displaystyle \forall x \in \mathbb{R}\ \left( \sqrt{2x - 1} = 2\sqrt{x + 4} - 3 \iff \left( x = 5 \right) \right)$
\item $\displaystyle \forall x \in \mathbb{R}\ \left( -\sqrt{x^{2} - 1} + \sqrt{x^{2} + 3x} = 2 \iff \left( x = 1 \right) \right)$
\item $\displaystyle \forall x \in \mathbb{R}\ \left( 2\sqrt{x + 5} - \sqrt{2x + 1} = \sqrt{3x - 3} \iff \left( x = 4 \right) \right)$
\item $\displaystyle \forall x \in \mathbb{R}\ \left( \sqrt{x + 3} = \sqrt{x + 12} - \sqrt{3 - x} \iff \left( x = -\frac{12}{5} \lor x = 0 \right) \right)$
\item $\displaystyle \forall x \in \mathbb{R}\ \left( \sqrt{3} - \sqrt{\frac{x}{4} + 1} = \frac{1}{2}\sqrt{2 - x} \iff \left( x = -1 \right) \right)$
\item $\displaystyle \forall x \in \mathbb{R}\ \left( \sqrt{3x + 13} - \sqrt{3(x + 2)} = \sqrt{x + 3} - \sqrt{x} \iff \left( x = 1 \right) \right)$
\item $\displaystyle \forall x \in \mathbb{R}\ \left( \frac{2}{\sqrt{4x + 5}} = \frac{9}{\sqrt{4x^{2} + x - 5}} - \frac{1}{\sqrt{x - 1}} \iff \left( x = 5 \right) \right)$
\item $\displaystyle \forall x \in \mathbb{R}\ \left( \sqrt{5 - x} + \sqrt{5 + x} = \frac{12}{\sqrt{5 + x}} \iff \left( x = 3 \lor x = 4 \right) \right)$
\end{enumerate}
