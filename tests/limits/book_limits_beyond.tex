\title{Limits the engine does not decide yet (12)}
\author{}
\date{}
\begin{document}
\maketitle
Every statement below is true, with the result printed in a collection of exercises and checked by an independent numerical oracle. The engine does not prove them yet; it must never call them false.

\begin{enumerate}
\item $\displaystyle \lim_{x \to +\infty} \frac{\exp(\sqrt{x})(\sqrt{x} - 1)}{\ln\left(2\sqrt{4x^{2} - 2x + 1} + 4x - 1\right)} = +\infty$
\item $\displaystyle \lim_{x \to 0^{+}} \frac{\ln x}{\sqrt{1 + 2\ln^{2} x}} = -\sqrt{\frac{1}{2}}$
\item $\displaystyle \nexists \lim_{x \to +\infty} \sqrt{x}\cos x$
\item $\displaystyle \lim_{x \to -\infty} (8x + x\sin^{2} x) = -\infty$
\item $\displaystyle \lim_{x \to 0^{+}} \left(1 + e^{1/x^{3}}\right)^{\sin x} = +\infty$
\item $\displaystyle \lim_{x \to +\infty} \frac{\ln x}{\ln\ln x} = +\infty$
\item $\displaystyle \lim_{x \to +\infty} 2^{-x}x^{x} = +\infty$
\item $\displaystyle \lim_{x \to +\infty} \left(3^{(x^{2})} - (3^{x})^{2}\right) = +\infty$
\item $\displaystyle \nexists \lim_{x \to 0} \frac{e^{1/x} + 3}{e^{1/x} + 1}$
\item $\displaystyle \lim_{x \to +\infty} \frac{e^{3x} + 1}{e^{2x} + 2} = +\infty$
\item $\displaystyle \lim_{x \to +\infty} \frac{x^{2} - 5x}{x + 1}\left(e - 2^{\sin x}\right) = +\infty$
\item $\displaystyle \nexists \lim_{x \to +\infty} \frac{x^{2} - 5x}{x + 1}\left(e - e^{\sin x}\right)$
\end{enumerate}
