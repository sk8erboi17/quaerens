Every statement below is true, like those of \texttt{book\_integrals.tex}, but the engine does not decide it yet: an improper integral, a function its rules of differentiation do not know ($\arcsin$, $\cosh$, $\operatorname{arcosh}$), an absolute value to split, a value it cannot reduce exactly. It may answer that it cannot decide; it must never call one of them false.

\begin{enumerate}
\item $\displaystyle \int \sqrt{x^{2} - 1} \, dx = \frac{x}{2}\sqrt{x^{2} - 1} - \frac{1}{2}\operatorname{arcosh} x + c$
\item $\displaystyle \int \sqrt{\frac{1 - x}{1 + x}} \, dx = \arcsin x + \sqrt{1 - x^{2}} + c$
\item $\displaystyle \int \frac{1}{\sqrt{e^{-2x} - 1}} \, dx = \arcsin\left(e^{x}\right) + c$
\item $\displaystyle \int_{-1}^{0} \sqrt[5]{x} \, dx = -\frac{5}{6}$
\item $\displaystyle \int_{1}^{2} \frac{1}{\sqrt{2 - x}} \, dx = 2$
\item $\displaystyle \int_{0}^{2} \frac{1}{\sqrt{4 - x^{2}}} \, dx = \frac{\pi}{2}$
\item $\displaystyle \int_{e}^{e^{\frac{3}{2}}} \frac{1}{x\left(1 - \sqrt{\ln x - 1}\right)} \, dx = -\sqrt{2} - 2\ln\left(1 - \frac{\sqrt{2}}{2}\right)$
\item $\displaystyle \int_{2}^{+\infty} (x - 1) e^{-x^{2} + 2x} \, dx = \frac{1}{2}$
\item $\displaystyle \int_{0}^{+\infty} \frac{\arctan\left(\sqrt{7x}\right)}{(1 + 7x)\sqrt{7x}} \, dx = \frac{\pi^{2}}{28}$
\item $\displaystyle \int_{1}^{+\infty} 3 \cdot \frac{\arctan x}{x^{2}} \, dx = 3 \cdot \left[\frac{\pi}{4} - \ln\frac{1}{\sqrt{2}}\right]$
\item $\displaystyle \int_{1}^{\infty} \frac{\ln x}{x^{2}} \, dx = 1$
\item $\displaystyle \int_{0}^{2} \frac{dx}{(2 - x)^{1/3}} = \frac{3}{2}2^{2/3}$
\item $\displaystyle \int_{0}^{1} \frac{x^{2}}{\sqrt{1 - x^{2}}} \, dx = \frac{\pi}{4}$
\item $\displaystyle \int_{0}^{\pi/2} \tan x \, dx = +\infty$
\item $\displaystyle \int_{0}^{\infty} \frac{dx}{\sqrt{x}(1 + x)} = \pi$
\item $\displaystyle \int_{1}^{\infty} \frac{1}{x} \, dx = +\infty$
\item $\displaystyle \int \frac{1 + \sqrt{x}}{x\left(1 + \sqrt[3]{x}\right)} \, dx = 6\sqrt[6]{x} + \ln x - 3\ln\left(1 + \sqrt[3]{x}\right) - 6\arctan\sqrt[6]{x} + c$
\item $\displaystyle \int \frac{dx}{\sqrt{x} + \sqrt[3]{x}} = 2\sqrt{x} - 3\sqrt[3]{x} + 6\sqrt[6]{x} - 6\ln\left(\sqrt[6]{x} + 1\right) + c$
\end{enumerate}
