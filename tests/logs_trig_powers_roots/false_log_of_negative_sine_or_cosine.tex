\title{Logarithms of sines and cosines that are not positive (12)}
\author{}
\date{}
\begin{document}
\maketitle
In each statement the two sides are the same expression, or follow from each other by plain algebra. But one of them takes the logarithm of a number that is negative, for every $n$ or for some $n$: $\cos 2$, $\sin 4$, $\sin n$ for $n = 4$, $\cos\frac{5}{2}$. So no statement holds for every $n \ge 1$.

\begin{enumerate}
\item $\displaystyle \ln\left(\cos 2\right) = \ln\left(\cos 2\right)$
\item $\displaystyle \sum_{k=1}^{n} \ln\left(\sin 4\right) = n \ln\left(\sin 4\right)$
\item $\displaystyle \ln\left(\sin n\right) = \ln\left(\sin n\right)$
\item $\displaystyle \ln\left(\cos n\right) = \ln\left(\cos n\right)$
\item $\displaystyle \sum_{k=1}^{n} \ln\left(\cos\left(\frac{5}{2^{k}}\right)\right) = \sum_{k=1}^{n} \ln\left(\cos\left(\frac{5}{2^{k}}\right)\right)$
\item $\displaystyle \sum_{k=1}^{n} \ln\left(\cos\left(\frac{x}{2^{k}}\right) - 2\right) = \sum_{k=1}^{n} \ln\left(\cos\left(\frac{x}{2^{k}}\right) - 2\right)$
\item $\displaystyle \ln\left(2^{n} \sin\left(-\frac{3}{2^{n}}\right)\right) = \ln\left(2^{n} \sin\left(-\frac{3}{2^{n}}\right)\right)$
\item $\displaystyle \ln\left(\tan 2\right) = \ln\left(\tan 2\right)$
\item $\displaystyle \ln\left(\cos\left(\frac{x}{2^{n}}\right)\right) + \ln\left(-\cos\left(\frac{x}{2^{n}}\right)\right) = \ln\left(\cos\left(\frac{x}{2^{n}}\right)\right) + \ln\left(-\cos\left(\frac{x}{2^{n}}\right)\right)$
\item $\displaystyle \log_{\cos 2}\left(4\right) = \log_{\cos 2}\left(4\right)$
\item $\displaystyle \ln\left(\frac{\sin 2}{\sin 4}\right) = \ln\left(\frac{\sin 2}{\sin 4}\right)$
\item $\displaystyle \sum_{k=1}^{n} \ln\left(\sin\left(\frac{4}{2^{k}}\right) \cos\left(\frac{4}{2^{k}}\right)\right) = \sum_{k=1}^{n} \ln\left(\sin\left(\frac{4}{2^{k}}\right) \cos\left(\frac{4}{2^{k}}\right)\right)$
\end{enumerate}
