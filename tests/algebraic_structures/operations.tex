$\mathbb{N} = \{1, 2, 3, \ldots\}$, $\mathbb{N}_{0} = \mathbb{N} \cup \{0\}$. An operation is declared among the hypotheses with its sets and its rule, as a map is; in the thesis it is written between its two arguments, with brackets whenever it is repeated.

\begin{enumerate}
\item $\displaystyle (\perp \colon \mathbb{Z} \times \mathbb{Z} \to \mathbb{Z} \land \forall a, b \in \mathbb{Z}\ (a \perp b = a - b)) \implies (\lnot (\forall a, b, c \in \mathbb{Z}\ ((a \perp b) \perp c = a \perp (b \perp c))) \land \lnot (\forall a, b \in \mathbb{Z}\ (a \perp b = b \perp a)) \land \exists e \in \mathbb{Z}\ \forall a \in \mathbb{Z}\ (a \perp e = a) \land \lnot (\exists e \in \mathbb{Z}\ \forall a \in \mathbb{Z}\ (e \perp a = a)) \land \forall a, x, y \in \mathbb{Z}\ ((a \perp x = a \perp y \implies x = y) \land (x \perp a = y \perp a \implies x = y)))$
\item $\displaystyle (\perp \colon \mathbb{Z} \times \mathbb{Z} \to \mathbb{Z} \land \forall a, b \in \mathbb{Z}\ (a \perp b = a^{2} + b)) \implies (\lnot (\forall a, b, c \in \mathbb{Z}\ ((a \perp b) \perp c = a \perp (b \perp c))) \land \lnot (\forall a, b \in \mathbb{Z}\ (a \perp b = b \perp a)) \land \exists e \in \mathbb{Z}\ \forall a \in \mathbb{Z}\ (e \perp a = a) \land \lnot (\exists e \in \mathbb{Z}\ \forall a \in \mathbb{Z}\ (a \perp e = a)) \land \forall a \in \mathbb{Z}\ \lnot (\forall x, y \in \mathbb{Z}\ ((a \perp x = a \perp y \implies x = y) \land (x \perp a = y \perp a \implies x = y))))$
\item $\displaystyle (\perp \colon \mathbb{Z} \times \mathbb{Z} \to \mathbb{Z} \land \forall a, b \in \mathbb{Z}\ (a \perp b = 4a + b)) \implies (\lnot (\forall a, b, c \in \mathbb{Z}\ ((a \perp b) \perp c = a \perp (b \perp c))) \land \lnot (\forall a, b \in \mathbb{Z}\ (a \perp b = b \perp a)) \land \exists e \in \mathbb{Z}\ \forall a \in \mathbb{Z}\ (e \perp a = a) \land \lnot (\exists e \in \mathbb{Z}\ \forall a \in \mathbb{Z}\ (a \perp e = a)) \land \forall a, x, y \in \mathbb{Z}\ ((a \perp x = a \perp y \implies x = y) \land (x \perp a = y \perp a \implies x = y)) \land \forall a, b \in \mathbb{N}\ \exists c \in \mathbb{N}\ (a \perp b = c))$
\item $\displaystyle (\perp \colon \mathbb{Z} \times \mathbb{Z} \to \mathbb{Z} \land \forall a, b \in \mathbb{Z}\ (a \perp b = a^{2} + b^{2})) \implies (\forall a, b \in \mathbb{Z}\ (a \perp b = b \perp a) \land \lnot (\forall a, b, c \in \mathbb{Z}\ ((a \perp b) \perp c = a \perp (b \perp c))) \land \forall a \in \mathbb{Z}\ \lnot (\forall x, y \in \mathbb{Z}\ ((a \perp x = a \perp y \implies x = y) \land (x \perp a = y \perp a \implies x = y))) \land \lnot (\exists e \in \mathbb{Z}\ \forall a \in \mathbb{Z}\ (e \perp a = a \land a \perp e = a)))$
\item $\displaystyle (\perp \colon \mathbb{Z} \times \mathbb{Z} \to \mathbb{Z} \land \forall a, b \in \mathbb{Z}\ (a \perp b = 3(a + b))) \implies (\forall a, b \in \mathbb{Z}\ (a \perp b = b \perp a) \land \lnot (\forall a, b, c \in \mathbb{Z}\ ((a \perp b) \perp c = a \perp (b \perp c))) \land \forall a, x, y \in \mathbb{Z}\ ((a \perp x = a \perp y \implies x = y) \land (x \perp a = y \perp a \implies x = y)) \land \lnot (\exists e \in \mathbb{Z}\ \forall a \in \mathbb{Z}\ (e \perp a = a \land a \perp e = a)))$
\item $\displaystyle (\perp \colon \mathbb{N} \times \mathbb{N} \to \mathbb{N} \land \forall a, b \in \mathbb{N}\ (a \perp b = a + b + ab)) \implies (\forall a, b, c \in \mathbb{N}\ ((a \perp b) \perp c = a \perp (b \perp c)) \land \forall a, b \in \mathbb{N}\ (a \perp b = b \perp a) \land \forall a, x, y \in \mathbb{N}\ ((a \perp x = a \perp y \implies x = y) \land (x \perp a = y \perp a \implies x = y)) \land \lnot (\exists e \in \mathbb{N}\ \forall a \in \mathbb{N}\ (e \perp a = a \land a \perp e = a)))$
\item $\displaystyle (\ast \colon \mathbb{Q} \times \mathbb{Q} \to \mathbb{Q} \land \forall a, b \in \mathbb{Q}\ (a \ast b = a + b - ab)) \implies (\forall a, b, c \in \mathbb{Q}\ ((a \ast b) \ast c = a \ast (b \ast c)) \land \forall a, b \in \mathbb{Q}\ (a \ast b = b \ast a) \land \forall a \in \mathbb{Q}\ (a \ast 0 = a \land 0 \ast a = a) \land \forall a \in \mathbb{Q}\ (a \ast 1 = 1))$

\item $\displaystyle (\circ \colon \mathbb{Q} \times \mathbb{Q} \to \mathbb{Q} \land \forall x, y \in \mathbb{Q}\ (x \circ y = \frac{x + y}{2})) \implies (\forall a, b \in \mathbb{Q}\ (a \circ b = b \circ a) \land \lnot (\forall a, b, c \in \mathbb{Q}\ ((a \circ b) \circ c = a \circ (b \circ c))) \land \forall a \in \mathbb{Q}\ (a \circ a = a) \land \lnot (\forall a \in \mathbb{Q}\ (0 \circ a = a)))$
\item $\displaystyle (\diamond \colon \mathbb{Z} \times \mathbb{N} \to \mathbb{Z} \land \forall x \in \mathbb{Z}\ \forall n \in \mathbb{N}\ (x \diamond n = nx)) \implies (\forall x \in \mathbb{Z}\ (x \diamond 1 = x) \land \forall x \in \mathbb{Z}\ \forall m, n \in \mathbb{N}\ ((x \diamond m) \diamond n = (x \diamond n) \diamond m) \land \lnot (\forall x \in \mathbb{Z}\ \forall m, n \in \mathbb{N}\ (x \diamond m = x \diamond n \implies m = n)))$
\item $\displaystyle (\oplus \colon \mathbb{N}_{0} \times \mathbb{N}_{0} \to \mathbb{N}_{0} \land \forall b, a \in \mathbb{N}_{0}\ (a \oplus b = a + 2b)) \implies (\lnot (\forall a, b \in \mathbb{N}_{0}\ (a \oplus b = b \oplus a)) \land \forall a \in \mathbb{N}_{0}\ (a \oplus 0 = a) \land \lnot (\forall a \in \mathbb{N}_{0}\ (0 \oplus a = a)) \land \forall a, x, y \in \mathbb{N}_{0}\ (a \oplus x = a \oplus y \implies x = y))$

\item $\displaystyle (f \colon \mathbb{N} \to \mathbb{Z} \land \forall n \in \mathbb{N}\ (f(n) = n - 5) \land g \colon \mathbb{Z} \to \mathbb{N}_{0} \land \forall n \in \mathbb{Z}\ (g(n) = n^{2})) \implies (\lnot (\forall a, b \in \mathbb{N}\ (g(f(a)) = g(f(b)) \implies a = b)) \land \forall a \in \mathbb{N}\ (g(f(a)) = a^{2} - 10a + 25))$
\item $\displaystyle (\ast \colon \mathbb{Z} \times \mathbb{Z} \to \mathbb{Z} \land \forall a, b \in \mathbb{Z}\ (a \ast b = ab + 1) \land f \colon \mathbb{Z} \to \mathbb{Z} \land \forall n \in \mathbb{Z}\ (f(n) = -n)) \implies (\forall a, b \in \mathbb{Z}\ (f(a) \ast f(b) = a \ast b) \land \lnot (\forall a, b \in \mathbb{Z}\ (f(a \ast b) = f(a) \ast f(b))))$
\end{enumerate}
